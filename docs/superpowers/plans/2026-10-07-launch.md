**Goal:** Ship and prove the build: the running `DEVEX_LOG.md`, the `pop-demo` personas and smoke checks, the Docker Compose deployment, the human-gated mainnet steps (VPS and S1, the analyst's ERC-8004 identity, smoke checks S2–S5 and the money-shot rehearsal), the judge-facing README, and the pre-public cleanup.
**Architecture:** `@pop/demo` is a small workspace package. Its caller persona drives the Trading and Transaction APIs through `@pop/core`, and its checks (feed health, the analyst's identity, DevEx evidence) are the gates of the human-gated tasks, so a person proves each mainnet step with one command. The deployment is one VPS running Caddy (TLS, static feed, `/api` and `/agent` routes), the API server with its keeper, and the analyst. Secrets exist only in git-ignored env files and the analyst's `.studio/` on that VPS.
**Tech Stack:** Node ≥ 22.12, pnpm 10.34.6, TypeScript ~5.9.3 (strict, ESM), Vitest ^3.2.7, tsx ^4.23.15, viem ^2.57.3, zod ^3.25.76, Docker Compose v2, Caddy 2, `node:22-bookworm-slim` images

/** The DX form's sections (event.md §3.2) plus our smoke checks; DEVEX_LOG.md keeps them in this order. */
export const DEVEX_SECTIONS = [
  "Onboarding",
  "Documentation issues",
  "API pitfalls",
  "Latency",
  "Rate limits",
  "Signing",
  "Data trust",
  "AI stack",
  "Tokenized-stock specifics",
  "Redesign ideas",
  "Requested capabilities",
  "Smoke evidence",
] as const;

export const SMOKE_SECTIONS = ["S1", "S2", "S3", "S4", "S5", "Rehearsal"] as const;

/** The bullet lines under one `##`/`###` heading, up to the next heading of any level. */
export function sectionBullets(text: string, heading: string): string[] | null {
  const lines = text.replace(/\r\n/g, "\n").split("\n");
  const start = lines.findIndex((line) => /^#{2,3} /.test(line) && line.replace(/^#{2,3} /, "") === heading);
  if (start === -1) return null;
  const bullets: string[] = [];
  for (const line of lines.slice(start + 1)) {
    if (/^#{1,3} /.test(line)) break;
    if (line.startsWith("- ")) bullets.push(line.slice(2));
  }
  return bullets;
}

/**
 * Problems with the log for the named sections: a missing heading, or no evidence yet (only the
 * template's "(pending)" bullet). An empty list means every named section has real evidence.
 */
export function devexProblems(text: string, sections: readonly string[]): string[] {
  const problems: string[] = [];
  for (const section of sections) {
    const bullets = sectionBullets(text, section);
    if (bullets === null) problems.push(`${section}: heading missing`);
    else if (!bullets.some((bullet) => !bullet.startsWith("(pending)") && bullet.trim().length > 10)) problems.push(`${section}: no evidence yet`);
  }
  return problems;
}

import { parseRpcUrls } from "@pop/core";

export interface DemoEnv {
  registry: `0x${string}` | null;
  rpcUrls: string[];
  w3w: { apiKey: string; apiSecret: string } | null;
  /** The caller persona's burner key (CALLER_PK). Never printed. */
  callerKey: `0x${string}` | null;
  telemetryFile: string | null;
  /** The deployed feed (POP_FEED_URL), e.g. https://<host>. */
  feedUrl: string | null;
  analystAgentId: bigint | null;
  analystWallet: `0x${string}` | null;
}

const ADDRESS = /^0x[0-9a-fA-F]{40}$/;

function optionalAddress(name: string, raw: string | undefined): `0x${string}` | null {
  const value = raw?.trim() ?? "";
  if (value === "") return null;
  if (!ADDRESS.test(value)) throw new Error(`${name} must be a 0x-prefixed 20-byte address`);
  return value as `0x${string}`;
}

/** Reads the demo environment (names only in docs: REGISTRY_ADDRESS, BSC_RPC_URLS, W3W_API_KEY/SECRET, CALLER_PK, …). */
export function loadDemoEnv(env: Record<string, string | undefined>): DemoEnv {
  const apiKey = env.W3W_API_KEY?.trim() ?? "";
  const apiSecret = env.W3W_API_SECRET?.trim() ?? "";
  if ((apiKey === "") !== (apiSecret === "")) throw new Error("set both W3W_API_KEY and W3W_API_SECRET, or neither");
  const callerRaw = env.CALLER_PK?.trim() ?? "";
  if (callerRaw !== "" && !/^0x[0-9a-fA-F]{64}$/.test(callerRaw)) throw new Error("CALLER_PK must be a 0x-prefixed 32-byte hex key");
  const agentRaw = env.ANALYST_AGENT_ID?.trim() ?? "";
  if (agentRaw !== "" && !/^\d+$/.test(agentRaw)) throw new Error("ANALYST_AGENT_ID must be the numeric ERC-8004 agent id");
  return {
    registry: optionalAddress("REGISTRY_ADDRESS", env.REGISTRY_ADDRESS),
    rpcUrls: parseRpcUrls(env.BSC_RPC_URLS),
    w3w: apiKey === "" ? null : { apiKey, apiSecret },
    callerKey: callerRaw === "" ? null : (callerRaw as `0x${string}`),
    telemetryFile: env.DEVEX_TELEMETRY_FILE?.trim() || null,
    feedUrl: env.POP_FEED_URL?.trim().replace(/\/$/, "") || null,
    analystAgentId: agentRaw === "" ? null : BigInt(agentRaw),
    analystWallet: optionalAddress("ANALYST_WALLET", env.ANALYST_WALLET),
  };
}

import {
  ADDRESSES,
  BSC_CHAIN_ID,
  CALL_REGISTRY_ABI,
  ERC20_ABI,
  broadcastTx,
  isSimulationSuccess,
  parseUnits18,
  sendWithTransactionApi,
  simulateTx,
  simulationFailure,
  swapWithTradingApi,
  type RegistryReader,
  type SwapClient,
  type TokenInfo,
  type TxSigner,
  type W3wClient,
} from "@pop/core";
import { encodeFunctionData, parseEventLogs } from "viem";

export interface CallerDeps {
  w3w: W3wClient;
  client: SwapClient & RegistryReader;
  signer: TxSigner;
  registry: `0x${string}`;
}

/** Gas limit for the deliberately reverting try-gate post (a revert cannot be estimated). */
export const TRY_GATE_GAS = 200_000n;

/** Buys `usdt` USDT of `token` (Trading API, simulate first, MEV protection on). */
export async function callerBuy(deps: CallerDeps, token: TokenInfo, usdt: string): Promise<{ received: bigint; txHash: `0x${string}` }> {
  const swap = await swapWithTradingApi({ ...deps, fromToken: ADDRESSES.usdt, toToken: token.address, amount: parseUnits18(usdt), mev: true });
  return { received: swap.received, txHash: swap.swap.txHash };
}

/** post(token, amount, note) from the caller; the simulation refuses ("No position, no call.") when it does not hold enough. */
export async function callerPost(deps: CallerDeps, token: TokenInfo, amount: string, note: string): Promise<{ callId: number; txHash: `0x${string}` }> {
  const data = encodeFunctionData({ abi: CALL_REGISTRY_ABI, functionName: "post", args: [token.address, parseUnits18(amount), note] });
  const result = await sendWithTransactionApi({ w3w: deps.w3w, client: deps.client, signer: deps.signer, tx: { to: deps.registry, data }, mev: false });
  const [event] = parseEventLogs({ abi: CALL_REGISTRY_ABI, eventName: "CallPosted", logs: result.receipt.logs });
  if (!event) throw new Error(`post mined in ${result.txHash} but emitted no CallPosted`);
  return { callId: Number(event.args.callId), txHash: result.txHash };
}

/** Sells the caller's whole balance of `token` back to USDT: the money shot's "the caller sold". */
export async function callerSell(deps: CallerDeps, token: TokenInfo): Promise<{ sold: bigint; received: bigint; txHash: `0x${string}` } | null> {
  const balance = await deps.client.readContract({ address: token.address, abi: ERC20_ABI, functionName: "balanceOf", args: [deps.signer.address] });
  if (balance === 0n) return null;
  const swap = await swapWithTradingApi({ ...deps, fromToken: token.address, toToken: ADDRESSES.usdt, amount: balance, mev: true });
  return { sold: balance, received: swap.received, txHash: swap.swap.txHash };
}

/**
 * The gate, shown two ways: the Transaction API simulate of post(token, amount) from a caller that does
 * not hold it (the revert reason, nothing signed), and with `broadcast` the same post signed and sent
 * anyway, so BscScan shows the failed transaction with "No position, no call.".
 */
export async function callerTryGate(
  deps: CallerDeps,
  token: TokenInfo,
  amount: string,
  broadcast: boolean,
): Promise<{ wouldSucceed: boolean; failReason: string | null; txHash: `0x${string}` | null }> {
  const data = encodeFunctionData({ abi: CALL_REGISTRY_ABI, functionName: "post", args: [token.address, parseUnits18(amount), "Try the gate"] });
  const simulation = await simulateTx(deps.w3w, { from: deps.signer.address, to: deps.registry, value: "0", data });
  if (isSimulationSuccess(simulation)) return { wouldSucceed: true, failReason: null, txHash: null };
  const failReason = simulationFailure(simulation);
  if (!broadcast) return { wouldSucceed: false, failReason, txHash: null };
  const nonce = await deps.client.getTransactionCount({ address: deps.signer.address, blockTag: "pending" });
  const gasPrice = await deps.client.getGasPrice();
  const signedTransaction = await deps.signer.signTransaction({ chainId: BSC_CHAIN_ID, nonce, to: deps.registry, data, value: 0n, gas: TRY_GATE_GAS, gasPrice });
  const { txHash } = await broadcastTx(deps.w3w, { signedTransaction, address: deps.signer.address, enableMevProtection: false });
  await deps.client.waitForTransactionReceipt({ hash: txHash as `0x${string}`, timeout: 90_000 });
  return { wouldSucceed: false, failReason, txHash: txHash as `0x${string}` };
}

          executionMode: "SWAP",

        const status = options.simulation ?? "SUCCESS";
        return status === "SUCCESS" ? { status } : { status: "FAILED", failReason: status };
      }
      case "/api/v1/dex/pre-transaction/broadcast-transaction":
        broadcasts += 1;
        return { orderId: String(broadcasts), txHash: `0x${broadcasts.toString(16).padStart(64, "0")}` };

        throw new Error(`unexpected ${call.path}`);

      const bought = `0x${tx.data.slice(10, 50)}` as `0x${string}`;
      signed.push(`swap:${bought}`);

    }

      signed.push(`raw:${tx.to.toLowerCase()}`);
      return [];
    }
    signed.push("approve");
    return [];
  }

      switch (request.functionName) {

        default:

      }

    getTransactionCount: async () => 3,
    getGasPrice: async () => 100_000_000n,
    estimateGas: async () => 100_000n,

      return { status: reverted ? "reverted" : "success", transactionHash: hash, blockNumber: 300n, logs } as unknown as TransactionReceipt;
    },

    address: CALLER,
    async signTransaction(tx) {
      lastTx = tx;
      return "0xf86c01";
    },
  };

/** `pop-demo api get <path> k=v …`: one signed GET, its outcome and latency (DevEx evidence for S1/S3). */
export async function apiGet(w3w: W3wClient, path: string, query: Record<string, string>, clock: () => number = () => performance.now()): Promise<string[]> {
  const started = clock();
  try {
    const data = await w3w.get(path, query, z.unknown());
    const ms = Math.round(clock() - started);
    return [`GET ${path} → code 0 in ${ms} ms`, JSON.stringify(data).slice(0, 600)];
  } catch (error) {
    const ms = Math.round(clock() - started);
    if (error instanceof W3wApiError) return [`GET ${path} → code ${error.code} (HTTP ${error.httpStatus}) in ${ms} ms: ${error.msg}`];
    throw error;
  }
}

}

function requireRegistry(env: DemoEnv): `0x${string}` {
  if (!env.registry) throw new Error("set REGISTRY_ADDRESS to the deployed CallRegistry");
  return env.registry;
}

async function caller(deps: CliDeps, demo: DemoEnv, action: string | undefined, values: Record<string, string | boolean | undefined>): Promise<number> {
  const token = requireTokenBySymbol(String(values.token ?? ""));
  const persona = { w3w: deps.w3w(demo), client: deps.chain(demo), signer: deps.signer(demo), registry: requireRegistry(demo) };
  deps.out(`caller ${persona.signer.address}`);
  switch (action) {
    case "buy": {
      const bought = await callerBuy(persona, token, String(values.usdt ?? ""));
      deps.out(`bought ${bought.received} raw ${token.symbol} in ${bought.txHash}`);
      return 0;
    }
    case "post": {
      const posted = await callerPost(persona, token, String(values.amount ?? ""), String(values.note ?? ""));
      deps.out(`posted call #${posted.callId} on ${token.symbol} in ${posted.txHash}`);
      return 0;
    }
    case "sell": {
      if (values.all !== true) throw new Error("caller sell needs --all (it sells the caller's whole balance)");
      const sold = await callerSell(persona, token);
      deps.out(sold ? `sold ${sold.sold} raw ${token.symbol} for ${sold.received} raw USDT in ${sold.txHash}` : `the caller holds no ${token.symbol}`);
      return 0;
    }
    case "try-gate": {
      const gate = await callerTryGate(persona, token, String(values.amount ?? ""), values.broadcast === true);
      if (gate.wouldSucceed) deps.out(`this caller holds ${token.symbol}: the post would succeed (use a caller without it)`);
      else deps.out(`simulate → ${gate.failReason}${gate.txHash ? ` · broadcast anyway: ${gate.txHash} (reverted on-chain)` : ""}`);
      return 0;
    }
    default:
      return usage(deps, `unknown caller action ${action ?? ""}`);
  }
}

function usage(deps: CliDeps, problem: string): number {
  deps.err(`error: ${problem}`);
  deps.err(USAGE);
  return 1;
}

export async function main(argv: readonly string[], overrides: Partial<CliDeps> = {}): Promise<number> {
  const deps: CliDeps = { ...defaultDeps(overrides.env ?? process.env), ...overrides };
  const [command, ...rest] = argv;
  if (command === undefined || command === "--help" || command === "help") {
    deps.out(USAGE);
    return 0;
  }
  let values: Record<string, string | boolean | undefined>;
  let positionals: string[];
  try {
    ({ values, positionals } = parseArgs({
      args: [...rest],
      allowPositionals: true,
      strict: true,
      options: {
        token: { type: "string" },
        usdt: { type: "string" },
        amount: { type: "string" },
        note: { type: "string" },
        all: { type: "boolean" },
        broadcast: { type: "boolean" },
        limit: { type: "string" },
        wallet: { type: "string" },
        sections: { type: "string" },
      },
    }));
  } catch (error) {
    return usage(deps, error instanceof Error ? error.message : String(error));
  }
  try {
    const demo = loadDemoEnv(deps.env);
    switch (command) {
      case "caller":
        return await caller(deps, demo, positionals[0], values);
      case "status":
        for (const line of await statusLines({ client: deps.chain(demo), registry: requireRegistry(demo) }, Number(values.limit ?? "10"))) deps.out(line);
        return 0;
      case "agent": {
        if (demo.analystAgentId === null) throw new Error("set ANALYST_AGENT_ID (and ANALYST_WALLET to check it)");
        const result = await agentCheck({ client: deps.chain(demo), registry: requireRegistry(demo), agentId: demo.analystAgentId, expectedWallet: demo.analystWallet });
        for (const line of result.lines) deps.out(line);
        deps.out(result.ok ? "PASS" : "FAIL");
        return result.ok ? 0 : 1;
      }
      case "health": {
        if (!demo.feedUrl) throw new Error("set POP_FEED_URL to the deployed feed, e.g. https://<host>");
        const result = await healthCheck(demo.feedUrl, deps.fetchImpl);
        for (const line of result.lines) deps.out(line);
        deps.out(result.ok ? "PASS" : "FAIL");
        return result.ok ? 0 : 1;
      }
      case "api": {
        const [verb, path, ...pairs] = positionals;
        if (verb !== "get" || !path?.startsWith("/api/")) return usage(deps, "api get <path starting with /api/> [key=value …]");
        const query = Object.fromEntries(pairs.map((pair) => pair.split("=", 2) as [string, string]));
        for (const line of await apiGet(deps.w3w(demo), path, query)) deps.out(line);
        return 0;
      }
      case "trades": {
        const wallet = String(values.wallet ?? "");
        if (!/^0x[0-9a-fA-F]{40}$/.test(wallet)) return usage(deps, "--wallet <0x address> is required");
        const token = requireTokenBySymbol(String(values.token ?? ""));
        for (const line of await tradesLines(deps.w3w(demo), token.address, wallet as `0x${string}`, deps.now())) deps.out(line);
        return 0;
      }
      case "balances": {
        const wallet = String(values.wallet ?? "");
        if (!/^0x[0-9a-fA-F]{40}$/.test(wallet)) return usage(deps, "--wallet <0x address> is required");
        for (const line of await balanceLines(deps.w3w(demo), wallet as `0x${string}`)) deps.out(line);
        return 0;
      }
      case "devex-check": {
        const sections = String(values.sections ?? "").split(",").map((s) => s.trim()).filter((s) => s !== "");
        if (sections.length === 0) return usage(deps, "--sections S1,S2,… is required");
        const problems = devexProblems(readFileSync(deps.devexLogPath, "utf8"), sections);
        for (const problem of problems) deps.out(`FAIL ${problem}`);
        deps.out(problems.length === 0 ? "PASS" : "FAIL");
        return problems.length === 0 ? 0 : 1;
      }
      default:
        return usage(deps, `unknown command ${command}`);
    }
  } catch (error) {
    deps.err(error instanceof Error ? error.message : String(error));
    return 1;
  }
}

## What it does

Stock tips are everywhere, and nobody can tell whether the person shouting "buy NVIDIA" owns it or is selling into your buying. Proof of Position makes holding the price of speaking. ## How it works

1. **Hold first.** `post(token, amount, note)` checks the caller's balance in the same transaction. If the balance is short, it reverts with "No position, no call.", the contract's only revert string. All 675 BSC tokenized stocks are listed: bStocks and Ondo are measured by `balanceOf`, and rebasing xStocks by non-rebasing `sharesOf`, so a dividend or a split can't flip a badge.
2. **A live badge.** Every browser re-reads the newest calls with one `getCalls` per BSC block (about 0.45 s) from public RPC, with no server in the path. HOLDING means the balance is at or above the declared amount, TRIMMED means below it, EXITED means zero. CLOSED marks a call the caller ended, either "held through" or "exited while open".
3. **Sticky exits.** `recordExit(callId)` is open to anyone. Our keeper sends it within a couple of blocks of a sale, and a buy-back can't erase it.
4. **Exit evidence.** The server finds the transaction where the tokens left: Binance Market API trades for the caller's wallet first, `Transfer` logs otherwise. The card says whether it was a sale.
5. **Agents.** `pop-follower` mirrors chosen callers on a Binance Agentic Wallet within a budget and sells when their badge flips; it ships as a Wallet Skill (`skills/pop-follower`). The analyst (Agent Studio) researches a bStock with RWA Data, the Market API and CoinMarketCap paid over x402, writes a thesis with a Pieverse model, buys, posts as its ERC-8004 identity, and closes after its hold period. The registry scores each of its calls in the ERC-8004 ReputationRegistry: 100 if held through, 50 if trimmed, 0 if exited.
6. **Every signed transaction is simulated first** with the Binance Transaction API, and a failed simulation is never signed.

Binance Web3 API modules used: RWA Data (`rwa/tokens`), Market (`trades`, `token/top-liquidity`), Trading (`quote`, `swap`, `approve-transaction`), Transaction (`simulate`, `broadcast-transaction`), Wallet (`token-balances-by-address`), plus Agentic Wallet (`baw`) and BNB Agent Studio.
