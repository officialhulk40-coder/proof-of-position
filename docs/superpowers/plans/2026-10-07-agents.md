# Agents Implementation Plan

**Goal:** Build the two agents: `@pop/follower` (`pop-follower`), which mirrors chosen callers on a Binance Agentic Wallet and exits when their badge flips, with its Wallet Skill; and `popanalyst`, a BNB Agent Studio agent that researches a bStock (RWA Data, Market API, CoinMarketCap via x402), writes a thesis with its Pieverse model, buys, posts the call on-chain as its ERC-8004 identity and closes it after its hold period.
**Architecture:** The follower is a pnpm workspace package: a pure `decide()` over registry reads, an executor that drives the `baw` CLI (submit, poll to FINISHED, measure the balance delta), JSON state, and a CLI. The analyst is the Agent Studio 0.0.14 scaffold (`bag init`, its own pnpm workspace under `agents/popanalyst/app/agent`) plus `src/analyst/*`, wired in with one hook line in `unifiedMain.ts`; it signs only through `@pop/core`'s simulate → sign → broadcast pipeline with the Studio wallet. Every network edge is injected, so both agents are tested with fakes and nothing in this plan's executor tasks spends money.
**Tech Stack:** Node ≥ 22.12, pnpm 10.34.6, TypeScript ~5.9.3 (strict, ESM), Vitest ^3.2.7, tsx ^4.23.15, viem ^2.57.3, zod ^3.25.76, Binance Agentic Wallet CLI `baw` ≥ 1.10.0 (runtime only), `@bnbagent/studio-cli@0.0.14` (`bag`), `@bnbagent/studio-runtime@0.0.14`, `@bnbagent/sdk@0.6.0`, AI SDK `ai` ^7.0.29
**Source spec:** docs/superpowers/specs/2026-10-06-proof-of-position-design.md

Execution notes (apply to every task in this plan):

- Run every command from the repository root. Each `Run:` line is one command that works unchanged in PowerShell and Git Bash. The one exception is the human-gated Task 12, whose commands a person runs from the directory it names.
- Follower imports are extensionless (`./baw`), like the rest of the root workspace. Analyst files live inside the Agent Studio scaffold, whose `tsconfig.json` is `NodeNext`, so their relative imports end in `.js` (`./config.js`). They are typechecked with `tsconfig.analyst.json` (Bundler resolution), which also covers `@pop/core`'s TypeScript source.

```ts
import { spawn } from "node:child_process";
import { z } from "zod";

/** One `baw` invocation's exit code and output. */
export interface BawResult {
  code: number;
  stdout: string;
  stderr: string;
}

/** Runs `baw <args>`. The real one spawns the CLI; tests pass a scripted fake. */
export type BawRunner = (args: readonly string[]) => Promise<BawResult>;

/** The Agentic Wallet CLI version this follower is written against (Wallet Skill 1.12.0). */
export const REQUIRED_BAW_VERSION = "1.10.0";

export class BawError extends Error {
  constructor(message: string) {
    super(message);
    this.name = "BawError";
  }
}

const SAFE_ARG = /^[A-Za-z0-9_.:-]+$/;

/**
 * The real runner. On Windows `baw` is an npm `.cmd` shim, which only starts through a shell, so every
 * argument is first checked against a strict pattern (addresses, numbers, flags): nothing can be injected.
 */
export function createBawRunner(options: { bin?: string; timeoutMs?: number } = {}): BawRunner {
  const bin = options.bin ?? "baw";
  const timeoutMs = options.timeoutMs ?? 60_000;
  return (args) =>
    new Promise((resolve, reject) => {
      const unsafe = args.find((arg) => !SAFE_ARG.test(arg));
      if (unsafe !== undefined) {
        reject(new BawError(`refusing to pass an unsafe argument to baw: ${unsafe}`));
        return;
      }
      const child = spawn(bin, [...args], { shell: process.platform === "win32", windowsHide: true });
      let stdout = "";
      let stderr = "";
      const timer = setTimeout(() => {
        child.kill();
        reject(new BawError(`baw ${args.slice(0, 2).join(" ")} timed out after ${timeoutMs} ms`));
      }, timeoutMs);
      child.stdout.on("data", (chunk: Buffer) => {
        stdout += chunk.toString("utf8");
      });
      child.stderr.on("data", (chunk: Buffer) => {
        stderr += chunk.toString("utf8");
      });
      child.on("error", (error) => {
        clearTimeout(timer);
        reject(new BawError(`could not start ${bin}: ${error.message} (install it: npm install -g @binance/agentic-wallet@${REQUIRED_BAW_VERSION})`));
      });
      child.on("close", (code) => {
        clearTimeout(timer);
        resolve({ code: code ?? 1, stdout, stderr });
      });
    });
}

const envelopeSchema = z.object({ success: z.boolean(), data: z.unknown().optional() }).passthrough();

/** The CLI's own error text, verbatim: error.message / message / msg, else the whole JSON. */
function failureText(envelope: Record<string, unknown>): string {
  const error = envelope.error;
  if (error && typeof error === "object" && typeof (error as { message?: unknown }).message === "string") return (error as { message: string }).message;
  if (typeof error === "string") return error;
  if (typeof envelope.message === "string") return envelope.message;
  if (typeof envelope.msg === "string") return envelope.msg;
  return JSON.stringify(envelope);
}

/** Runs `baw <args> --json`, requires `{ success: true }` and parses `data` with `schema`. */
export async function bawJson<T>(runner: BawRunner, args: readonly string[], schema: z.ZodType<T, z.ZodTypeDef, unknown>): Promise<T> {
  const command = `baw ${args.join(" ")}`;
  const result = await runner([...args, "--json"]);
  let parsed: unknown;
  try {
    parsed = JSON.parse(result.stdout.trim());
  } catch {
    throw new BawError(`${command} printed no JSON (exit ${result.code}): ${(result.stderr || result.stdout).trim().slice(0, 400)}`);
  }
  const envelope = envelopeSchema.safeParse(parsed);
  if (!envelope.success) throw new BawError(`${command} printed an unexpected JSON shape: ${result.stdout.trim().slice(0, 400)}`);
  if (!envelope.data.success) throw new BawError(`${command} failed: ${failureText(envelope.data)}`);
  return schema.parse(envelope.data.data);
}

const orderIdSchema = z.union([z.string(), z.number()]).transform((value) => String(value));

export async function cliCheck(runner: BawRunner, requiredVersion = REQUIRED_BAW_VERSION): Promise<{ currentCliVersion: string; needUpdateCli: boolean }> {
  return bawJson(runner, ["cli-check", "--required-version", requiredVersion], z.object({ currentCliVersion: z.string(), needUpdateCli: z.boolean() }));
}

/** `UNCONNECTED` (signed out), `CREATING` or `CONNECTED`. */
export async function walletStatus(runner: BawRunner): Promise<string> {
  return (await bawJson(runner, ["wallet", "status"], z.object({ status: z.string() }))).status;
}

/** The wallet's BSC (binanceChainId "56") address, lower-cased. */
export async function bscAddress(runner: BawRunner): Promise<`0x${string}`> {
  const data = await bawJson(runner, ["wallet", "address"], z.object({ addresses: z.array(z.object({ binanceChainId: z.string(), address: z.string() })) }));
  const bsc = data.addresses.find((entry) => entry.binanceChainId === "56");
  if (!bsc || !/^0x[0-9a-fA-F]{40}$/.test(bsc.address)) throw new BawError("baw wallet address returned no BSC (56) address");
  return bsc.address.toLowerCase() as `0x${string}`;
}

export const walletSettingsSchema = z
  .object({
    sessionExpireTime: z.string().nullish(),
    inactiveSignOutTime: z.string().nullish(),
    dailyLimit: z.number().nullish(),
    quotaLeft: z.number().nullish(),
    tradeAllTokens: z.boolean().nullish(),
    abnormalTxnHandling: z.string().nullish(),
  })
  .passthrough();
export type WalletSettings = z.infer<typeof walletSettingsSchema>;

export async function walletSettings(runner: BawRunner): Promise<WalletSettings> {
  return bawJson(runner, ["wallet", "settings"], walletSettingsSchema);
}

/** `LOCKED` while a tx is pending or a double-confirm waits in the Binance App; else `UNLOCKED`. */
export async function txLock(runner: BawRunner): Promise<string> {
  return (await bawJson(runner, ["wallet", "tx-lock", "--binanceChainId", "56"], z.object({ status: z.string() }))).status;
}

/** Submits a BSC market order and returns its orderId (submitted, not filled: poll getMarketOrder). */
export async function marketSwap(
  runner: BawRunner,
  params: { fromTokenQty: string; fromToken: string; toToken: string; slippage?: string; mev?: boolean },
): Promise<string> {
  const args = [
    "market-order",
    "swap",
    "--fromTokenQty",
    params.fromTokenQty,
    "--fromToken",
    params.fromToken,
    "--toToken",
    params.toToken,
    "--binanceChainId",
    "56",
    "--slippage",
    params.slippage ?? "auto",
    "--mev",
    String(params.mev ?? true),
  ];
  return (await bawJson(runner, args, z.object({ orderId: orderIdSchema }))).orderId;
}

export const marketOrderSchema = z
  .object({
    orderId: orderIdSchema,
    status: z.string(),
    txHash: z.string().nullish(),
    fromTokenQty: z.string().nullish(),
    toToken: z.string().nullish(),
  })
  .passthrough();
export type MarketOrder = z.infer<typeof marketOrderSchema>;

/** `baw market-order list --orderId <id>`: PENDING, FINISHED or FAILED (txHash may be null on FAILED). */
export async function getMarketOrder(runner: BawRunner, orderId: string): Promise<MarketOrder | null> {
  const data = await bawJson(runner, ["market-order", "list", "--orderId", orderId], z.object({ list: z.array(marketOrderSchema) }));
  return data.list.find((order) => order.orderId === orderId) ?? null;
}
```

```ts
/** A CallView: live status 0 HOLDING, 1 TRIMMED, 2 EXITED, 3 CLOSED. */
export function call(id: number, overrides: Partial<RawCall> = {}, status = 0): CallView {
  const raw: RawCall = {
    caller: CALLER,
    token: NVDAB,
    model: 1,
    hasAgent: false,

    exitStatus: 0,
    postedBlock: 100n,
    postedAt: 1_791_000_000n,
    closedBlock: 0n,
    exitBlock: 0n,
    agentId: 0n,
    declaredAmount: 20_000_000_000_000_000n,
    declaredBase: 20_000_000_000_000_000n,
    note: "",
    ...overrides,

/**
 * A scripted `baw`: handlers are keyed by the command words ("wallet status", "market-order swap",
 * "cli-check"); each returns `data` (wrapped as { success: true, data }) or a raw BawResult.
 */
export function scriptedBaw(handlers: Record<string, (args: readonly string[]) => unknown>): { runner: BawRunner; calls: string[][] } {
  const calls: string[][] = [];
  const runner: BawRunner = async (args) => {
    calls.push([...args]);
    const words = args[1] !== undefined && !args[1].startsWith("--") ? `${args[0]} ${args[1]}` : String(args[0]);
    const handler = handlers[words];
    if (!handler) return { code: 1, stdout: JSON.stringify({ success: false, error: { message: `unscripted: ${words}` } }), stderr: "" };
    const answer = handler(args);
    if (answer && typeof answer === "object" && "stdout" in answer) return answer as BawResult;
    return { code: 0, stdout: JSON.stringify({ success: true, data: answer }), stderr: "" };
  };
  return { runner, calls };
}

/** A manual clock: sleep() advances it instantly. */
export function manualClock(start = 1_000_000) {
  let now = start;
  return {
    now: () => now,
    sleep: async (ms: number) => {
      now += ms;
    },
  };
}
```

Spec §8.1. `decide` is pure. SELL every open mirror whose call is now TRIMMED, EXITED or CLOSED, or has a recorded exit (`exitBlock != 0`). Then BUY, oldest first, each call that is HOLDING, open, with no recorded exit, posted by a followed caller on a **bStock** (`isFollowable`), with an id at or above the start cursor and no mirror yet, while fewer than `maxOpen` mirrors are open. Sells come first. A failed mirror stays in state, so a failing call is never retried forever. State writes go to a temp file and are renamed, and a corrupt state file is an error, never a silent reset.

```ts
import { getTokenInfo, isFollowable, type CallView } from "@pop/core";

export interface FollowerConfig {
  /** Callers to mirror (any case). */
  callers: readonly string[];
  /** USDT spent on each mirrored call, decimal string (e.g. "1"). */
  budgetUsdt: string;
  /** Most mirrors held at once. */
  maxOpen: number;
}

export type MirrorStatus = "open" | "closed" | "failed";

/** One mirrored call: what was bought, and how it ended. */
export interface Mirror {
  callId: number;
  caller: string;
  token: string;
  status: MirrorStatus;
  /** Raw token amount bought (and the exact amount to sell), as a decimal string. */
  quantity: string;
  buyOrderId: string | null;
  buyTxHash: string | null;
  sellOrderId: string | null;
  sellTxHash: string | null;
  openedAt: string | null;
  closedAt: string | null;
  /** TRIMMED, EXITED, CLOSED or EXIT_RECORDED once sold. */
  exitReason: string | null;
  error: string | null;
}

export interface FollowerState {
  version: 1;
  /** Calls with a lower id were posted before the follower started; they are never bought. */
  cursor: number;
  /** Keyed by call id. A failed mirror stays here so the same call is never retried forever. */
  mirrors: Record<string, Mirror>;
}

export type Action = { kind: "BUY"; call: CallView } | { kind: "SELL"; call: CallView; mirror: Mirror; reason: string };

/** Why a mirrored call must be sold now, or null while the caller still holds through. */
export function exitReason(call: CallView): string | null {
  if (call.status === "TRIMMED" || call.status === "EXITED" || call.status === "CLOSED") return call.status;
  if (call.exitBlock > 0n) return "EXIT_RECORDED";
  return null;
}

/**
 * Pure decision step, run on every registry read. SELL every open mirror whose call flipped
 * (TRIMMED, EXITED, CLOSED, or a recorded exit); then BUY each new HOLDING call by a followed caller on
 * a bStock, oldest first, while fewer than `maxOpen` mirrors are open. Sells come first.
 */
export function decide(calls: readonly CallView[], state: FollowerState, config: FollowerConfig): Action[] {
  const byId = new Map(calls.map((call) => [call.id, call]));
  const actions: Action[] = [];
  for (const mirror of Object.values(state.mirrors)) {
    if (mirror.status !== "open") continue;
    const call = byId.get(mirror.callId);
    const reason = call ? exitReason(call) : null;
    if (call && reason) actions.push({ kind: "SELL", call, mirror, reason });
  }

  const followed = new Set(config.callers.map((caller) => caller.toLowerCase()));
  let open = Object.values(state.mirrors).filter((mirror) => mirror.status === "open").length;
  for (const call of [...calls].sort((a, b) => a.id - b.id)) {
    if (open >= config.maxOpen) break;
    if (call.id < state.cursor || state.mirrors[String(call.id)] !== undefined) continue;
    if (!followed.has(call.caller.toLowerCase())) continue;
    if (call.status !== "HOLDING" || call.closedBlock !== 0n || call.exitBlock !== 0n) continue;
    const token = getTokenInfo(call.token);
    if (!token || !isFollowable(token)) continue;
    actions.push({ kind: "BUY", call });
    open += 1;
  }
  return actions;
}
```

```ts
import { getMarketOrder, marketSwap, txLock, walletStatus, type BawRunner } from "./baw";

export interface ExecutorDeps {
  baw: BawRunner;
  /** Raw ERC-20 balance read from BSC (viem readContract balanceOf). */
  balanceOf: (token: `0x${string}`, owner: `0x${string}`) => Promise<bigint>;
  /** The Agentic Wallet's BSC address. */
  wallet: `0x${string}`;
  now: () => number;
  sleep: (ms: number) => Promise<void>;
  log: (line: string) => void;
  /** Order and lock polling interval. Default 2 s. */
  pollMs?: number;
  /** Cap on waiting for a lock to clear or an order to finish. Default 90 s. */
  timeoutMs?: number;
}

/** The Agentic Wallet is signed out: stop the loop and ask the user to sign in again. */
export class WalletSignedOutError extends Error {
  constructor(status: string) {
    super(`Agentic Wallet is ${status}: sign in again with \`baw auth signin\`, then restart pop-follower`);
    this.name = "WalletSignedOutError";
  }
}

export class OrderFailedError extends Error {
  constructor(message: string) {
    super(message);
    this.name = "OrderFailedError";
  }
}

export interface OrderOutcome {
  orderId: string;
  txHash: string | null;
  submittedAt: number;
  finishedAt: number;
}

/** Throws WalletSignedOutError unless `baw wallet status` is CONNECTED. */
export async function requireConnected(baw: BawRunner): Promise<void> {
  const status = await walletStatus(baw);
  if (status !== "CONNECTED") throw new WalletSignedOutError(status);
}

/** Waits while the wallet is LOCKED (a pending tx, or a double-confirm waiting in the Binance App). */
async function waitUnlocked(deps: ExecutorDeps): Promise<void> {
  const deadline = deps.now() + (deps.timeoutMs ?? 90_000);
  for (;;) {
    if ((await txLock(deps.baw)) !== "LOCKED") return;
    if (deps.now() >= deadline) throw new OrderFailedError("the wallet stayed LOCKED (pending tx or Binance App confirmation) for 90 s");
    deps.log("wallet LOCKED (a pending tx, or a confirmation waiting in the Binance App); waiting");
    await deps.sleep(deps.pollMs ?? 2_000);
  }
}

/** Submits one market order and polls it to FINISHED or FAILED: a submitted order is not a fill. */
async function runOrder(deps: ExecutorDeps, order: { fromTokenQty: string; fromToken: string; toToken: string }): Promise<OrderOutcome & { status: string }> {
  await waitUnlocked(deps);
  const submittedAt = deps.now();
  const orderId = await marketSwap(deps.baw, { ...order, slippage: "auto", mev: true });
  const deadline = submittedAt + (deps.timeoutMs ?? 90_000);
  for (;;) {
    await deps.sleep(deps.pollMs ?? 2_000);
    const state = await getMarketOrder(deps.baw, orderId);
    if (state && (state.status === "FINISHED" || state.status === "FAILED")) {
      return { orderId, txHash: state.txHash ?? null, status: state.status, submittedAt, finishedAt: deps.now() };
    }
    if (deps.now() >= deadline) throw new OrderFailedError(`order ${orderId} is still ${state?.status ?? "unknown"} after 90 s`);
  }
}

/** A FAILED order (slippage, liquidity, routing) is logged and retried once. */
async function runOrderWithRetry(deps: ExecutorDeps, order: { fromTokenQty: string; fromToken: string; toToken: string }, label: string): Promise<OrderOutcome> {
  const first = await runOrder(deps, order);
  if (first.status === "FINISHED") return first;
  deps.log(`${label}: order ${first.orderId} FAILED; retrying once`);
  const second = await runOrder(deps, order);
  if (second.status === "FINISHED") return second;
  throw new OrderFailedError(`${label}: order ${second.orderId} FAILED twice`);
}

/** Buys `budgetUsdt` USDT of `token`; the quantity is the wallet's on-chain balance increase. */
export async function executeBuy(deps: ExecutorDeps, params: { token: `0x${string}`; budgetUsdt: string }): Promise<OrderOutcome & { quantity: bigint }> {
  const before = await deps.balanceOf(params.token, deps.wallet);
  const outcome = await runOrderWithRetry(deps, { fromTokenQty: params.budgetUsdt, fromToken: ADDRESSES.usdt, toToken: params.token }, "BUY");
  // FINISHED means mined, but a public RPC can lag a block: re-read briefly until the balance moves.
  let after = await deps.balanceOf(params.token, deps.wallet);
  for (let attempt = 0; attempt < 5 && after <= before; attempt += 1) {
    await deps.sleep(1_000);
    after = await deps.balanceOf(params.token, deps.wallet);
  }
  if (after <= before) throw new OrderFailedError(`BUY order ${outcome.orderId} finished but the token balance did not increase`);
  return { ...outcome, quantity: after - before };
}

/** Sells exactly `quantity` (raw units) of `token` back to USDT: only what the follower bought. */
export async function executeSell(deps: ExecutorDeps, params: { token: `0x${string}`; quantity: bigint }): Promise<OrderOutcome> {
  return runOrderWithRetry(deps, { fromTokenQty: formatUnits18(params.quantity, 18), fromToken: params.token, toToken: ADDRESSES.usdt }, "SELL");
}
```

```ts
import { formatUnits18, getTokenInfo, readCallCount, readCallsFrom, type CallView, type RegistryReader, type TelemetryEvent } from "@pop/core";
import { decide, type Action, type FollowerConfig, type FollowerState, type Mirror } from "./decide";
import { executeBuy, executeSell, requireConnected, WalletSignedOutError, type ExecutorDeps } from "./executor";
import { loadState, saveState } from "./state";

export interface FollowerDeps {
  chain: RegistryReader;
  registry: `0x${string}`;
  executor: ExecutorDeps;
  config: FollowerConfig;
  statePath: string;
  log: (line: string) => void;
  /** DevEx telemetry (JSONL); optional. */
  record?: (event: TelemetryEvent) => void;
}

export interface Follower {
  readonly state: FollowerState;
  /** One pass: read the registry, decide, execute sequentially, persist. Returns what it did. */
  tick(): Promise<Action[]>;
}

function symbolOf(token: string): string {
  return getTokenInfo(token)?.symbol ?? token;
}

function errorText(error: unknown): string {
  return error instanceof Error ? error.message : String(error);
}

/**
 * Loads (or starts) the state. On first start the cursor is the current call count, so only calls
 * posted from now on are mirrored; `includeExisting` starts it at 0 instead.
 */
export async function createFollower(deps: FollowerDeps, options: { includeExisting?: boolean } = {}): Promise<Follower> {
  let state = loadState(deps.statePath);
  if (state === null) {
    const cursor = options.includeExisting ? 0 : await readCallCount(deps.chain, deps.registry);
    state = { version: 1, cursor, mirrors: {} };
    saveState(deps.statePath, state);
    deps.log(`pop-follower: new state at ${deps.statePath}; mirroring calls from #${cursor}`);
  }
  const current: FollowerState = state;
  let running = false;

  async function buy(call: CallView, detectedAt: number): Promise<void> {
    const symbol = symbolOf(call.token);
    deps.log(`Call #${call.id}: ${call.caller} is HOLDING ${symbol} — buying ${deps.config.budgetUsdt} USDT of it`);
    const base: Mirror = {
      callId: call.id,
      caller: call.caller,
      token: call.token,
      status: "failed",
      quantity: "0",
      buyOrderId: null,
      buyTxHash: null,
      sellOrderId: null,
      sellTxHash: null,
      openedAt: null,
      closedAt: null,
      exitReason: null,
      error: null,
    };
    try {
      const result = await executeBuy(deps.executor, { token: call.token, budgetUsdt: deps.config.budgetUsdt });
      current.mirrors[String(call.id)] = {
        ...base,
        status: "open",
        quantity: result.quantity.toString(),
        buyOrderId: result.orderId,
        buyTxHash: result.txHash,
        openedAt: new Date(result.finishedAt).toISOString(),
      };
      deps.log(`Mirrored call #${call.id}: you hold ${formatUnits18(result.quantity, 6)} ${symbol} alongside them (order ${result.orderId}, ${((result.finishedAt - detectedAt) / 1000).toFixed(1)} s)`);
      deps.record?.({
        kind: "follower.buy",
        callId: call.id,
        token: call.token,
        orderId: result.orderId,
        txHash: result.txHash,
        detectToSubmitMs: result.submittedAt - detectedAt,
        submitToFinishMs: result.finishedAt - result.submittedAt,
      });
    } catch (error) {
      if (error instanceof WalletSignedOutError) throw error;
      current.mirrors[String(call.id)] = { ...base, error: errorText(error) };
      deps.log(`BUY for call #${call.id} failed: ${errorText(error)}`);
    }
  }

  async function sell(call: CallView, mirror: Mirror, reason: string, detectedAt: number): Promise<void> {
    const symbol = symbolOf(mirror.token);
    const quantity = BigInt(mirror.quantity);
    deps.log(`Call #${call.id} is ${reason} — selling the mirrored ${formatUnits18(quantity, 6)} ${symbol}`);
    try {
      const result = await executeSell(deps.executor, { token: mirror.token as `0x${string}`, quantity });
      Object.assign(mirror, {
        status: "closed",
        sellOrderId: result.orderId,
        sellTxHash: result.txHash,
        closedAt: new Date(result.finishedAt).toISOString(),
        exitReason: reason,
        error: null,
      } satisfies Partial<Mirror>);
      deps.log(`Caller exited — sold ${formatUnits18(quantity, 6)} ${symbol} in ${((result.finishedAt - detectedAt) / 1000).toFixed(1)} s (order ${result.orderId}, tx ${result.txHash ?? "n/a"})`);
      deps.record?.({
        kind: "follower.sell",
        callId: call.id,
        reason,
        exitBlock: call.exitBlock.toString(),
        orderId: result.orderId,
        txHash: result.txHash,
        detectToSubmitMs: result.submittedAt - detectedAt,
        submitToFinishMs: result.finishedAt - result.submittedAt,
      });
    } catch (error) {
      if (error instanceof WalletSignedOutError) throw error;
      Object.assign(mirror, { status: "failed", error: errorText(error) } satisfies Partial<Mirror>);
      deps.log(`SELL for call #${call.id} failed: ${errorText(error)} — sell ${symbol} manually`);
    }
  }

  return {
    get state() {
      return current;
    },
    async tick() {
      if (running) return [];
      running = true;
      try {
        const openIds = Object.values(current.mirrors)
          .filter((mirror) => mirror.status === "open")
          .map((mirror) => mirror.callId);
        const calls = await readCallsFrom(deps.chain, deps.registry, Math.min(current.cursor, ...openIds));
        const actions = decide(calls, current, deps.config);
        if (actions.length === 0) return [];
        await requireConnected(deps.executor.baw);
        for (const action of actions) {
          const detectedAt = deps.executor.now();
          if (action.kind === "SELL") await sell(action.call, action.mirror, action.reason, detectedAt);
          else await buy(action.call, detectedAt);
          saveState(deps.statePath, current);
        }
        return actions;
      } finally {
        running = false;
      }
    },
  };
}
```

```ts
    { includeExisting: values["include-existing"] === true },
  );
  deps.out(`following ${callers.join(", ")} with ${budget} USDT per call (max ${maxOpen} open) from ${wallet}`);
  do {
    try {
      await follower.tick();
    } catch (error) {
      if (error instanceof WalletSignedOutError) {
        deps.err(error.message);
        return 2;
      }
      deps.err(`tick failed: ${error instanceof Error ? error.message : String(error)}`);
    }
    if (values.once === true) break;
    await deps.sleep(1_000);
  } while (deps.keepRunning());
  return 0;
}

function status(deps: CliDeps): number {
  const path = defaultStatePath(deps.env);
  const state = loadState(path);
  if (state === null) {
    deps.out(`no state yet at ${path}`);
    return 0;
  }
  deps.out(`state ${path} · cursor #${state.cursor}`);
  for (const mirror of Object.values(state.mirrors).sort((a, b) => a.callId - b.callId)) {
    const detail = mirror.status === "failed" ? `error: ${mirror.error}` : mirror.status === "closed" ? `sold on ${mirror.exitReason} (order ${mirror.sellOrderId})` : `holding (order ${mirror.buyOrderId})`;
    deps.out(`#${mirror.callId} ${mirror.token} ${formatUnits18(BigInt(mirror.quantity), 6)} · ${mirror.status} · ${detail}`);
  }
  return 0;
}

/** S4 smoke: buy then sell a few cents of a bStock with the Agentic Wallet; prints latencies. Moves funds only with --yes. */
async function smoke(deps: CliDeps, values: Record<string, string | boolean | string[] | undefined>): Promise<number> {
  const symbol = String(values.token ?? "");
  const usdt = String(values.usdt ?? "");
  if (!DECIMAL.test(usdt) || Number(usdt) <= 0) return usage(deps, "--usdt <n> must be a positive amount");
  const token = requireTokenBySymbol(symbol);
  if (!isFollowable(token)) return usage(deps, `${symbol} is not a bStock`);
  if (values.yes !== true) {
    deps.out(`would buy ${usdt} USDT of ${token.symbol} (${token.address}) with the Agentic Wallet, then sell exactly what arrived; add --yes to do it`);
    return 0;
  }
  const wallet = await bscAddress(deps.baw);
  const executor: ExecutorDeps = { baw: deps.baw, balanceOf: balanceReader(deps.chain), wallet, now: deps.now, sleep: deps.sleep, log: deps.out };
  const bought = await executeBuy(executor, { token: token.address, budgetUsdt: usdt });
  deps.out(`BUY ${formatUnits18(bought.quantity, 6)} ${token.symbol}: order ${bought.orderId}, tx ${bought.txHash ?? "n/a"}, ${bought.finishedAt - bought.submittedAt} ms submit→FINISHED`);
  const sold = await executeSell(executor, { token: token.address, quantity: bought.quantity });
  deps.out(`SELL ${formatUnits18(bought.quantity, 6)} ${token.symbol}: order ${sold.orderId}, tx ${sold.txHash ?? "n/a"}, ${sold.finishedAt - sold.submittedAt} ms submit→FINISHED`);
  return 0;
}

function usage(deps: CliDeps, problem?: string): number {
  if (problem) deps.err(`error: ${problem}`);
  deps.err(USAGE);
  return 1;
}

export async function main(argv: readonly string[], overrides: Partial<CliDeps> = {}): Promise<number> {
  const deps: CliDeps = { ...defaultDeps(overrides.env ?? process.env), ...overrides };
  const [command, ...rest] = argv;
  if (command === undefined || command === "--help" || command === "help") {
    deps.out(USAGE);
    return 0;
  }
  let values: Record<string, string | boolean | string[] | undefined>;
  try {
    ({ values } = parseArgs({
      args: [...rest],
      options: {
        caller: { type: "string", multiple: true },
        "budget-usdt": { type: "string" },
        "max-open": { type: "string" },
        "include-existing": { type: "boolean" },
        once: { type: "boolean" },
        token: { type: "string" },
        usdt: { type: "string" },
        yes: { type: "boolean" },
      },
      strict: true,
    }));
  } catch (error) {
    return usage(deps, error instanceof Error ? error.message : String(error));
  }
  try {
    switch (command) {
      case "check":
        return await check(deps);
      case "run":
        return await run(deps, values);
      case "status":
        return status(deps);
      case "smoke":
        return await smoke(deps, values);
      default:
        return usage(deps, `unknown command ${command}`);
    }
  } catch (error) {
    deps.err(error instanceof Error ? error.message : String(error));
    return 1;
  }
}
```

```markdown
# Proof of Position follower

Proof of Position is a feed of stock calls that can only be posted by someone who holds the
tokenized stock on BSC. Each call's badge (HOLDING, TRIMMED, EXITED, CLOSED) is read on-chain from
the caller's live balance by the CallRegistry contract. `pop-follower` turns that badge into
orders on the user's Binance Agentic Wallet:

- when a followed caller posts a HOLDING call on a **bStock**, it buys `--budget-usdt` USDT of it;
- when that call's badge flips (TRIMMED, EXITED, CLOSED, or an exit recorded on-chain), it sells

It drives the `baw` CLI (Agentic Wallet 1.10.0+) and reads the registry from public BSC RPC.

1. Run `pop-follower check`. It verifies the `baw` version, that the wallet is CONNECTED, how long
```

```markdown
# pop-follower policy

What the follower does, and what it never does.

## It acts only on the registry

- Its only signal is the CallRegistry's status for each call, read from BSC: HOLDING, TRIMMED,
  EXITED, CLOSED, and whether an exit was recorded. No prices, no sentiment, no other feeds.
- It follows only the callers named with `--caller`, and only calls on **bStocks** (Ondo trades only
  by RFQ, and xStocks have too little BSC liquidity).
- On first start it mirrors only calls posted from then on, unless `--include-existing` is given.

## It never exceeds the budget

- Each mirrored call spends exactly `--budget-usdt` USDT.
- At most `--max-open` mirrors are open at once.
- A buy that fails twice is marked failed and never retried for that call.
- The Agentic Wallet's own limits (daily quota, allowed tokens, abnormal-transaction handling) still
  apply on top, and the follower cannot change them.

## It sells only what it bought

- The quantity of a mirror is the wallet's on-chain balance increase from that buy.
- On a flip it sells exactly that quantity, never the rest of the wallet's holding.
- A sell that fails twice is marked failed and reported: the user sells by hand.

## The only commands it runs

`baw cli-check`, `baw wallet status`, `baw wallet address`, `baw wallet settings`,
`baw wallet tx-lock`, `baw market-order swap` and `baw market-order list`. Every order is a BSC
market order with `--slippage auto` and MEV protection on.

## Authorization

Starting `pop-follower run` is the user's standing authorization to trade within the stated budget
until they stop it (Ctrl+C). A signed-out wallet stops the loop with "sign in again". State lives in
`POP_FOLLOWER_STATE` (default `~/.pop-follower/state.json`), so a restart resumes the same mirrors.
```

```ts
 * x402 micropay buyer tools — emitted by `bag recipe code x402-buyer`.
 *
 * The agent's LLM can buy from any x402-protected URL whose host is on the
 * effective allowlist: `[payments.x402].allowed_hosts` plus every merchant
 * trusted via `bag x402 trust` (`[payments.x402.merchants.*]` — e.g. the
 * Binance Bazaar shelf: CMC market data).
 *
 * Wire into the agent (main.ts / mcpMain.ts on agentcore; foundryMain.ts on
 * azure-foundry — each carries this hint on its `tools:` line) — signing
 * stays fixed code, these tools only *fetch with payment* inside the config
 * fences:
 *
 *     import { X402_BUYER_TOOLS } from "./x402Buyer.js";
 *     ...
 *     tools: { ...LLM_READ_TOOLS, ...X402_BUYER_TOOLS },
 *
 * (This file sits NEXT TO tools.ts on purpose so the emitted agent package
 * stays a flat module set.)
 *
 * Security spine (all enforced in @bnbagent/studio-runtime, not here):
 *   - recipient: the pinned merchant `pay_to` (or `expected_recipient`) is
 *     byte-compared against the 402's payTo — a tampered 402 can never
 *     redirect funds;
 *   - per-call cap: `merchants.<name>.per_call_cap_usd` clamps every payment
 *     regardless of the `max_usd` argument below (the LLM can tighten it,
 *     never widen);
 *   - daily cap: `[budget].max_per_day_usd` across all paid calls.
 *
 * You own this file; edit freely.
 */

import { loadStudioToml } from "@bnbagent/studio-runtime/config";
import { getWallet } from "@bnbagent/studio-runtime/wallet";
import {
  fetchWithPayment,
  quoteUrl,
  X402BuyerPolicy,
  X402Error,
} from "@bnbagent/studio-runtime/x402";
import { tool, type ToolSet } from "ai";
import { z } from "zod";

function policy(): X402BuyerPolicy {
  return X402BuyerPolicy.fromToml(loadStudioToml());
}

/**
 * Pay an x402-protected endpoint and return its response.
 *
 * NO API KEY IS REQUIRED. x402 merchants (CoinMarketCap x402 endpoints,
 * Binance Bazaar merchants, etc.) authenticate the caller by the on-chain
 * EIP-3009 payment signed with the agent's own wallet — there is NO
 * `Authorization` / `X-API-KEY` header. If a task needs data from a URL on
 * the allowlist (CMC prices, etc.), ALWAYS use this tool; never ask the user
 * for an API key and never answer that the task can't be done for lack of
 * one. Payment is fenced by studio.toml (merchant.per_call_cap_usd +
 * [budget].max_per_day_usd), so calling it is always safe.
 *
 * URL host must be on the effective allowlist (allowed_hosts + trusted
 * merchants). `maxUsd` is only a refusal threshold — the actual price comes
 * from the endpoint's 402 challenge, and the config's per-call / daily caps
 * still apply on top. Query-style endpoints (e.g. CMC) take GET with params
 * in the URL; RPC-style ones take POST.
 */
export async function buyWithX402(
  url: string,
  maxUsd: number,
  method = "GET",
): Promise<Record<string, unknown>> {
  const pol = policy();
  try {
    const result = await fetchWithPayment(url, {
      maxUsd: Math.min(maxUsd, pol.maxPerRequestUsd),
      wallet: getWallet(),
      method,
      allowedHosts: [...pol.effectiveAllowedHosts],
    });
    return {
      ok: true,
      status: result.statusCode,
      json: result.json,
      paid_usd: result.paidUsd,
    };
  } catch (exc) {
    if (exc instanceof X402Error) {
      return { ok: false, error: exc.constructor.name, message: exc.message };
    }
    throw exc;
  }
}

/** Probe an x402 endpoint without paying — returns its 402 body for inspection. */
export async function quoteX402(
  url: string,
  method = "GET",
): Promise<Record<string, unknown>> {
  try {
    return {
      ok: true,
      accepts: await quoteUrl(url, {
        method,
        allowedHosts: [...policy().effectiveAllowedHosts],
      }),
    };
  } catch (exc) {
    if (exc instanceof X402Error) {
      return { ok: false, error: exc.constructor.name, message: exc.message };
    }
    throw exc;
  }
}

/** The AI SDK tool set — spread into the agent's `tools:` map. */
export const X402_BUYER_TOOLS: ToolSet = {
```

```ts
import { DEFAULT_WATCHLIST, getTokenBySymbol, isFollowable, parseRpcUrls } from "@pop/core";

/** Spend cap per call: ANALYST_BUY_USDT above this is refused at startup. */
export const MAX_BUY_USDT = 5;

export interface AnalystConfig {
  /** ANALYST_ENABLED=true starts the loop; anything else leaves the agent a plain Studio seller. */
  enabled: boolean;
  registry: `0x${string}` | null;
  /** ERC-8004 agent id from `bag erc8004 register`; null until registered (the loop then stays idle). */
  agentId: bigint | null;
  /** bStock symbols, in priority order. */
  watchlist: string[];
  /** USDT per new call, decimal string. */
  buyUsdt: string;
  holdHours: number;
  intervalMinutes: number;
  rpcUrls: string[];
  w3w: { apiKey: string; apiSecret: string } | null;
  /** Public feed URL used in the latest-call face (e.g. https://<host>). */
  feedUrl: string | null;
  telemetryFile: string | null;
}

function positiveNumber(name: string, raw: string | undefined, fallback: number, min: number): number {
  if (raw === undefined || raw.trim() === "") return fallback;
  const value = Number(raw);
  if (!Number.isFinite(value) || value < min) throw new Error(`${name} must be a number ≥ ${min}`);
  return value;
}

/** Reads the analyst's environment (names in spec §14; values live in the VPS env file, never in git). */
export function loadAnalystConfig(env: Record<string, string | undefined>): AnalystConfig {
  const registryRaw = env.REGISTRY_ADDRESS?.trim() ?? "";
  if (registryRaw !== "" && !/^0x[0-9a-fA-F]{40}$/.test(registryRaw)) throw new Error("REGISTRY_ADDRESS must be a 0x-prefixed 20-byte address");
  const agentRaw = env.ANALYST_AGENT_ID?.trim() ?? "";
  if (agentRaw !== "" && !/^\d+$/.test(agentRaw)) throw new Error("ANALYST_AGENT_ID must be the numeric ERC-8004 agent id");

  const watchlist = (env.ANALYST_WATCHLIST?.trim() ? env.ANALYST_WATCHLIST.split(",") : [...DEFAULT_WATCHLIST]).map((s) => s.trim()).filter((s) => s !== "");
  for (const symbol of watchlist) {
    const token = getTokenBySymbol(symbol);
    if (!token || !isFollowable(token)) throw new Error(`ANALYST_WATCHLIST: ${symbol} is not a bStock in the catalog`);
  }
  if (watchlist.length === 0) throw new Error("ANALYST_WATCHLIST is empty");

  const buyUsdt = env.ANALYST_BUY_USDT?.trim() || "2";
  if (!/^\d+(\.\d+)?$/.test(buyUsdt) || Number(buyUsdt) <= 0 || Number(buyUsdt) > MAX_BUY_USDT) {
    throw new Error(`ANALYST_BUY_USDT must be a positive amount of at most ${MAX_BUY_USDT}`);
  }

  const apiKey = env.W3W_API_KEY?.trim() ?? "";
  const apiSecret = env.W3W_API_SECRET?.trim() ?? "";
  if ((apiKey === "") !== (apiSecret === "")) throw new Error("set both W3W_API_KEY and W3W_API_SECRET, or neither");

  return {
    enabled: env.ANALYST_ENABLED?.trim().toLowerCase() === "true",
    registry: registryRaw === "" ? null : (registryRaw as `0x${string}`),
    agentId: agentRaw === "" ? null : BigInt(agentRaw),
    watchlist,
    buyUsdt,
    holdHours: positiveNumber("ANALYST_HOLD_HOURS", env.ANALYST_HOLD_HOURS, 24, 0.05),
    intervalMinutes: positiveNumber("ANALYST_INTERVAL_MIN", env.ANALYST_INTERVAL_MIN, 30, 1),
    rpcUrls: parseRpcUrls(env.BSC_RPC_URLS),
    w3w: apiKey === "" ? null : { apiKey, apiSecret },
    feedUrl: env.POP_FEED_URL?.trim().replace(/\/$/, "") || null,
    telemetryFile: env.DEVEX_TELEMETRY_FILE?.trim() || null,
  };
}

/** Why the loop cannot start with this config, or null when it can. */
export function loopBlocker(config: AnalystConfig): string | null {
  if (!config.enabled) return "ANALYST_ENABLED is not true";
  if (config.registry === null) return "REGISTRY_ADDRESS is not set";
  if (config.agentId === null) return "ANALYST_AGENT_ID is not set (register the agent on ERC-8004 first)";
  if (config.w3w === null) return "W3W_API_KEY / W3W_API_SECRET are not set";
  return null;
}
```

```ts
export function pickNumbers(json: unknown, keys: readonly string[]): Record<string, number> {
  const found: Record<string, number> = {};
  const queue: unknown[] = [json];
  while (queue.length > 0) {
    const node = queue.shift();
    if (Array.isArray(node)) queue.push(...node);
    else if (node !== null && typeof node === "object") {
      for (const [key, value] of Object.entries(node)) {
        if (keys.includes(key) && found[key] === undefined) {
          const number = typeof value === "number" ? value : typeof value === "string" ? Number(value) : Number.NaN;
          if (Number.isFinite(number)) found[key] = number;
        }
        if (value !== null && typeof value === "object") queue.push(value);
      }
    }
  }
```

```ts
import type { Research } from "./research.js";

/** CallRegistry's note limit (MAX_NOTE_BYTES); longer notes revert NoteTooLong. */
export const MAX_NOTE_BYTES = 280;

/** One LLM completion (the scaffold's buildModel() through the AI SDK's generateText). */
export type GenerateText = (prompt: string) => Promise<string>;

/** Collapses whitespace and trims to at most 280 UTF-8 bytes, ending with "…" when cut. */
export function clampNote(text: string): string {
  const flat = text.replace(/\s+/g, " ").replace(/^["'\s]+|["'\s]+$/g, "");
  const encoder = new TextEncoder();
  if (encoder.encode(flat).length <= MAX_NOTE_BYTES) return flat;
  let cut = "";
  for (const char of flat) {
    if (encoder.encode(`${cut}${char}…`).length > MAX_NOTE_BYTES) break;
    cut += char;
  }
  return `${cut.trimEnd()}…`;
}

function facts(research: Research): string {
  const lines = [`${research.ticker} via ${research.symbol} (bStocks on BSC)`];
  if (research.name) lines.push(`company: ${research.name}`);
  if (research.priceUsd) lines.push(`token price: $${research.priceUsd}`);
  if (research.market) lines.push(`market: ${research.market.openState ? "open" : "closed"} (${research.market.marketStatus}${research.market.reasonMsg ? `, ${research.market.reasonMsg}` : ""})`);
  if (research.pool) lines.push(`deepest pool: ${research.pool.protocol ?? "DEX"} with $${research.pool.liquidityUsd} liquidity`);
  if (research.cmc) lines.push(`CoinMarketCap pair quote: ${research.cmc.summary}`);
  return lines.join("\n");
}

/** The fallback when the LLM is unavailable: facts only, no hype. */
export function templateThesis(research: Research, holdHours: number): string {
  const price = research.priceUsd ? ` at $${research.priceUsd}` : "";
  const pool = research.pool ? `; deepest pool ${research.pool.protocol ?? "DEX"} ($${research.pool.liquidityUsd} liquidity)` : "";
  return clampNote(`Long ${research.ticker} (${research.symbol})${price}${pool}. Holding ~${holdHours}h; if I sell early this badge says so on-chain.`);
}

/** An LLM thesis grounded in the research, or the template on any failure. Never longer than 280 bytes. */
export async function writeThesis(generate: GenerateText | null, research: Research, holdHours: number): Promise<{ text: string; source: "llm" | "template" }> {
  if (generate !== null) {
    try {
      const prompt =
        `Write a one-sentence long thesis (at most 220 characters) for buying ${research.ticker} now, for a public call feed. ` +
        `Use only these facts:\n${facts(research)}\n` +
        `Plain text: no hashtags, no emojis, no price targets, no claims of certainty, and do not call it investment advice.`;
      const text = clampNote(await generate(prompt));
      if (text.length >= 20) return { text, source: "llm" };
    } catch {
      // fall through to the template
    }
  }
  return { text: templateThesis(research, holdHours), source: "template" };
}
```

```ts
import { CALL_REGISTRY_ABI, sendWithTransactionApi, type PipelineClient, type TxSigner, type W3wClient } from "@pop/core";
import { encodeFunctionData, parseEventLogs } from "viem";

export interface RegistryWriteDeps {
  w3w: W3wClient;
  client: PipelineClient;
  signer: TxSigner;
  registry: `0x${string}`;
}

/**
 * postAsAgent(token, amount, agentId, note) through the Transaction API pipeline (simulate first: a
 * revert such as "No position, no call." or NotAgentWallet signs nothing). Returns the new call id.
 */
export async function postAsAgent(
  deps: RegistryWriteDeps,
  params: { token: `0x${string}`; amount: bigint; agentId: bigint; note: string },
): Promise<{ callId: number; txHash: `0x${string}` }> {
  const data = encodeFunctionData({ abi: CALL_REGISTRY_ABI, functionName: "postAsAgent", args: [params.token, params.amount, params.agentId, params.note] });
  const result = await sendWithTransactionApi({ w3w: deps.w3w, client: deps.client, signer: deps.signer, tx: { to: deps.registry, data }, mev: false });
  const [event] = parseEventLogs({ abi: CALL_REGISTRY_ABI, eventName: "CallPosted", logs: result.receipt.logs });
  if (!event) throw new Error(`postAsAgent mined in ${result.txHash} but emitted no CallPosted`);
  return { callId: Number(event.args.callId), txHash: result.txHash };
}

/** close(callId): records any pending exit first, then writes the integrity feedback (100 when held through). */
export async function closeCall(deps: RegistryWriteDeps, callId: number): Promise<{ txHash: `0x${string}`; heldThrough: boolean | null }> {
  const data = encodeFunctionData({ abi: CALL_REGISTRY_ABI, functionName: "close", args: [BigInt(callId)] });
  const result = await sendWithTransactionApi({ w3w: deps.w3w, client: deps.client, signer: deps.signer, tx: { to: deps.registry, data }, mev: false });
  const [event] = parseEventLogs({ abi: CALL_REGISTRY_ABI, eventName: "CallClosed", logs: result.receipt.logs });
  return { txHash: result.txHash, heldThrough: event ? event.args.heldThrough : null };
}
```

```ts
  requireTokenBySymbol,

import { closeCall, postAsAgent } from "./registry.js";
import type { Research } from "./research.js";

/** Trading API codes that mean "not tradable right now": bStock RFQ closed (40369), no liquidity (40374). */
const NOT_TRADABLE_NOW = new Set([40369, 40374]);

export interface CycleDeps {
  config: AnalystConfig & { registry: `0x${string}`; agentId: bigint };
  w3w: W3wClient;
  client: SwapClient & RegistryReader;
  signer: TxSigner;
  research: (token: TokenInfo) => Promise<Research>;
  thesis: (research: Research) => Promise<{ text: string; source: "llm" | "template" }>;
  /** Milliseconds since the epoch. */
  now: () => number;
  log: (line: string) => void;
  record?: (event: TelemetryEvent) => void;
}

export interface CycleResult {
  closed: number[];
  posted: { callId: number; symbol: string; txHash: string } | null;
  skipped: string[];
}

function errorText(error: unknown): string {
  return error instanceof Error ? error.message : String(error);
}

/**
 * One analyst cycle (spec §8.2): (1) close own calls older than holdHours — close first, so the
 * registry records held-through honestly — then sell those positions; (2) for the first watchlist
 * bStock without an open call that can be bought now: research, thesis, buy, postAsAgent with the
 * amount actually received. At most one new call per cycle; every signature follows a simulation.
 */
export async function runCycle(deps: CycleDeps): Promise<CycleResult> {
  const { config } = deps;
  const me = deps.signer.address.toLowerCase();
  const writer = { w3w: deps.w3w, client: deps.client, signer: deps.signer, registry: config.registry };
  const mine = (await readCallsFrom(deps.client, config.registry, 0)).filter((call) => call.caller.toLowerCase() === me);
  const open = mine.filter((call) => call.closedBlock === 0n);
  const result: CycleResult = { closed: [], posted: null, skipped: [] };

  const holdSeconds = config.holdHours * 3_600;
  for (const call of open.filter((c: CallView) => deps.now() / 1_000 - c.postedAt >= holdSeconds)) {
    const closed = await closeCall(writer, call.id);
    result.closed.push(call.id);
    deps.log(`closed call #${call.id} (${closed.heldThrough === true ? "held through ✓" : "exited while open"}) in ${closed.txHash}`);
    const balance = await deps.client.readContract({ address: call.token, abi: ERC20_ABI, functionName: "balanceOf", args: [deps.signer.address] });
    if (balance > 0n) {
      const sold = await swapWithTradingApi({ w3w: deps.w3w, client: deps.client, signer: deps.signer, fromToken: call.token, toToken: ADDRESSES.usdt, amount: balance, mev: true });
      deps.log(`sold ${formatUnits18(balance, 6)} of ${call.token} for ${formatUnits18(sold.received, 4)} USDT in ${sold.swap.txHash}`);
    }
    deps.record?.({ kind: "analyst.close", callId: call.id, heldThrough: closed.heldThrough, txHash: closed.txHash });
  }

  const stillOpen = new Set(open.filter((call) => !result.closed.includes(call.id)).map((call) => call.token.toLowerCase()));
  for (const symbol of config.watchlist) {
    const token = requireTokenBySymbol(symbol);
    if (stillOpen.has(token.address)) {
      result.skipped.push(`${symbol}: a call is already open`);
      continue;
    }
    const research = await deps.research(token);
    const thesis = await deps.thesis(research);
    let bought;
    try {
      bought = await swapWithTradingApi({ w3w: deps.w3w, client: deps.client, signer: deps.signer, fromToken: ADDRESSES.usdt, toToken: token.address, amount: parseUnits18(config.buyUsdt), mev: true });
    } catch (error) {
      if (error instanceof NoSwapRouteError || (error instanceof W3wApiError && NOT_TRADABLE_NOW.has(error.code))) {
        result.skipped.push(`${symbol}: ${errorText(error)}`);
        continue;
      }
      throw error;
    }
    const posted = await postAsAgent(writer, { token: token.address, amount: bought.received, agentId: config.agentId, note: thesis.text });
    result.posted = { callId: posted.callId, symbol, txHash: posted.txHash };
    deps.log(`posted call #${posted.callId}: LONG ${symbol} ${formatUnits18(bought.received, 6)} (thesis by ${thesis.source}) in ${posted.txHash}`);
    deps.record?.({
      kind: "analyst.post",
      callId: posted.callId,
      symbol,
      amount: bought.received.toString(),
      thesisSource: thesis.source,
      cmcPaidUsd: research.cmc?.paidUsd ?? null,
      buyTxHash: bought.swap.txHash,
      postTxHash: posted.txHash,
    });
    return result;
  }
  return result;
}
```

```ts
export async function latestCall(deps: { client: RegistryReader; registry: `0x${string}`; wallet: string; feedUrl: string | null }): Promise<LatestCall | null> {
  const mine = (await readCallsFrom(deps.client, deps.registry, 0)).filter((call) => call.caller.toLowerCase() === deps.wallet.toLowerCase());
  const last = mine[mine.length - 1];
  if (!last) return null;
  return {
    callId: last.id,
    token: last.token,
    ticker: getTokenInfo(last.token)?.ticker ?? last.token,
    status: last.status,
    thesis: last.note,
    bscscan: `https://bscscan.com/token/${last.token}?a=${last.caller}`,
    feed: deps.feedUrl ? `${deps.feedUrl}/#/calls/${last.id}` : null,
  };
}

/** The production summarizer: this agent's own wallet, the env's registry and RPCs. Never throws. */
async function studioLatestCall(): Promise<unknown> {
  try {
    const config = loadAnalystConfig(process.env);
    if (config.registry === null) return { error: "REGISTRY_ADDRESS is not set" };
    const { getWallet } = await import("@bnbagent/studio-runtime/wallet");
    const wallet = getWallet().address;
    if (!wallet) return { error: "the agent wallet has no address yet" };
    const latest = await latestCall({ client: createBscPublicClient(config.rpcUrls), registry: config.registry, wallet, feedUrl: config.feedUrl });
    return latest ?? { message: "no calls yet" };
  } catch (error) {
    return { error: error instanceof Error ? error.message : String(error) };
  }
}

 * Wraps the scaffold's runWork (A2A and the FREE x402 route): a prompt starting with "latest" gets the
 * agent's latest call as JSON without touching the LLM; every other prompt falls through unchanged.
 */
export function withLatestCallFace(inner: RunWork, summarize: () => Promise<unknown> = studioLatestCall): RunWork {
  return async (prompt, opts) => (LATEST_PROMPT.test(prompt) ? JSON.stringify(await summarize()) : inner(prompt, opts));
```

```ts
export interface LoopHandle {
  stop(): void;
}

export interface Timers {
  setTimeout: (fn: () => void, ms: number) => unknown;
  setInterval: (fn: () => void, ms: number) => unknown;
  clearTimeout: (handle: unknown) => void;
  clearInterval: (handle: unknown) => void;
}

const realTimers: Timers = {
  setTimeout: (fn, ms) => setTimeout(fn, ms),
  setInterval: (fn, ms) => setInterval(fn, ms),
  clearTimeout: (handle) => clearTimeout(handle as ReturnType<typeof setTimeout>),
  clearInterval: (handle) => clearInterval(handle as ReturnType<typeof setInterval>),
};

/** First run after `firstDelayMs`, then every `intervalMs`; a cycle still running makes the next one a no-op. Never throws. */
export function scheduleCycles(run: () => Promise<unknown>, options: { intervalMs: number; firstDelayMs: number; log: (line: string) => void; timers?: Timers }): LoopHandle {
  const timers = options.timers ?? realTimers;
  let running = false;
  const tick = async () => {
    if (running) {
      options.log("previous cycle still running; skipping this one");
      return;
    }
    running = true;
    try {
      await run();
    } catch (error) {
      options.log(`cycle failed: ${error instanceof Error ? error.message : String(error)}`);
    } finally {
      running = false;
    }
  };
  const first = timers.setTimeout(() => void tick(), options.firstDelayMs);
  const every = timers.setInterval(() => void tick(), options.intervalMs);
  return {
    stop() {
      timers.clearTimeout(first);
      timers.clearInterval(every);
    },
  };
}

/** The agent's own wallet (`getWallet()`, unlocked with WALLET_PASSWORD) as a pipeline signer. The key never leaves it. */
export function studioSigner(): TxSigner {
  const wallet = getWallet();
  if (!wallet.address) throw new Error("the agent wallet has no address: run `bag wallet new` first");
  return walletProviderSigner({ address: wallet.address as `0x${string}`, signTransaction: (tx) => wallet.signTransaction(tx) });
}

/**
 * Called once from unifiedMain.ts main(). Starts the research → thesis → buy → post loop when
 * ANALYST_ENABLED=true and the registry, agent id and API keys are set; otherwise logs why and returns null.
 */
export function startAnalystLoop(env: Record<string, string | undefined> = process.env, log: (line: string) => void = (line) => console.log(`[analyst] ${line}`)): LoopHandle | null {
  try {
    const config = loadAnalystConfig(env);
    const blocker = loopBlocker(config);
    if (blocker !== null || config.registry === null || config.agentId === null || config.w3w === null) {
      log(`loop not started: ${blocker ?? "incomplete config"}`);
      return null;
    }
    const onTiming = config.telemetryFile ? createTelemetryRecorder({ filePath: config.telemetryFile, service: "analyst" }) : undefined;
    const w3w = createW3wClient({ apiKey: config.w3w.apiKey, apiSecret: config.w3w.apiSecret, onTiming });
    let model: ReturnType<typeof buildModel> | undefined;
    const generate = async (prompt: string) => {
      model ??= buildModel();
      return (await generateText({ model, prompt })).text;
    };
    const deps: CycleDeps = {
      config: { ...config, registry: config.registry, agentId: config.agentId },
      w3w,
      client: createBscPublicClient(config.rpcUrls),
      signer: studioSigner(),
      research: (token) => researchToken({ w3w, buyWithX402 }, token),
      thesis: (research) => writeThesis(generate, research, config.holdHours),
      now: () => Date.now(),
      log,
      record: config.telemetryFile ? createEventRecorder({ filePath: config.telemetryFile, service: "analyst" }) : undefined,
    };
    log(`loop started: every ${config.intervalMinutes} min, ${config.buyUsdt} USDT per call, hold ${config.holdHours} h, watchlist ${config.watchlist.join(",")}`);
    return scheduleCycles(
      async () => {
        const result = await runCycle(deps);
        log(`cycle done: closed [${result.closed.join(",")}], posted ${result.posted ? `#${result.posted.callId} ${result.posted.symbol}` : "nothing"}${result.skipped.length > 0 ? `, skipped ${result.skipped.join("; ")}` : ""}`);
      },
      { intervalMs: config.intervalMinutes * 60_000, firstDelayMs: 15_000, log },
    );
  } catch (error) {
    log(`loop not started: ${error instanceof Error ? error.message : String(error)}`);
    return null;
  }
}
```

```js
/** The key = value lines of one [section] (comments and other sections ignored). */
function section(text, name) {
  const values = {};
  let inside = false;
  for (const raw of text.replace(/\r\n/g, "\n").split("\n")) {
    const line = raw.replace(/\s+#.*$/, "").trim();
    const header = /^\[([^\]]+)\]$/.exec(line);
    if (header) {
      inside = header[1] === name;
      continue;
    }
    const pair = /^([A-Za-z0-9_]+)\s*=\s*(.+)$/.exec(line);
    if (inside && pair) values[pair[1]] = pair[2].replace(/^"(.*)"$/, "$1");
  }
  return values;
}
```
