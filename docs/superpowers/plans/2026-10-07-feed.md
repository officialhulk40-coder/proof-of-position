# Feed Implementation Plan

**Goal:** Build the feed: `@pop/server` (Hono on Node: token catalog with RWA market status, Transaction API dry-runs, Follow quote/build, exit evidence, and the `recordExit` keeper) and `@pop/web` (React 19 + Vite 7: live badges read from BSC on every block, "Try the gate", Post, Follow, alerts).
**Architecture:** The badge path never touches the server: the browser re-reads `getCalls` from public BSC RPC on every new block and diffs the result to detect flips. The server only enriches (market status, exit evidence), simulates (Transaction API), builds Follow transactions (Trading API) and runs the keeper. Every network edge is injected (`ServerDeps`, `Services`, the wallet `discover` function), so each route, hook and component is tested against fakes shaped like the real responses.
**Tech Stack:** Node ≥ 22.12, pnpm 10.34.6, TypeScript ~5.9.3 (strict, ESM, `moduleResolution: Bundler`), Hono ^4.13.13 + @hono/node-server ^1.19.17, React ^19.3.0, Vite ^7.3.7, @vitejs/plugin-react ^5.2.0, Vitest ^3.2.7 + jsdom ^26.1.0 + Testing Library, viem ^2.57.3, zod ^3.25.76
**Source spec:** docs/superpowers/specs/2026-10-06-proof-of-position-design.md

Execution notes (apply to every task in this plan):

- Run every command from the repository root. Each `Run:` line is one command that works unchanged in PowerShell and Git Bash.
- Imports are extensionless relative paths (`./config`), as `moduleResolution: Bundler` allows. Never use `as any`; when a test fake needs a cast, use `as unknown as T`.
- The server imports `@pop/core` (Node entry). The web imports only `@pop/core/browser` (no secrets, no `node:` modules).
- No test touches the network: the server tests use `createFakeW3wClient` and fake chain/log clients; the web tests use `fakeFetch`, `fakeChain` and a fake EIP-1193 provider.
- Wallet signing is never automatic: the web only calls `eth_sendTransaction` after a successful simulation and a user click. Nothing in this plan broadcasts a transaction.
- Cross-plan dependencies are written `Task <subsystem>:<N>`. `pnpm-lock.yaml` changes are an allowed side effect of any `pnpm install`.

```ts
import { DEFAULT_LOGS_RPC_URL, parseRpcUrls } from "@pop/core";
import { z } from "zod";

export interface ServerConfig {
  port: number;
  registry: `0x${string}`;
  rpcUrls: string[];
  logsRpcUrl: string;
  /** null → the W3W-backed routes answer 503 W3W_NOT_CONFIGURED; badges never need the API. */
  w3w: { apiKey: string; apiSecret: string } | null;
  /** Burner key for the recordExit keeper; null disables the keeper. Never logged. */
  keeperKey: `0x${string}` | null;
  telemetryFile: string | null;
  corsOrigins: string[];
}

const addressSchema = z
  .string({ required_error: "REGISTRY_ADDRESS is required" })
  .regex(/^0x[0-9a-fA-F]{40}$/, "REGISTRY_ADDRESS must be a 0x-prefixed 20-byte address");
const keySchema = z.string().regex(/^0x[0-9a-fA-F]{64}$/, "KEEPER_PK must be a 0x-prefixed 32-byte hex key");

/** Reads the server's environment (names only in docs; values come from the VPS env file). */
export function loadConfig(env: Record<string, string | undefined>): ServerConfig {
  const registry = addressSchema.parse(env.REGISTRY_ADDRESS) as `0x${string}`;
  const port = env.PORT === undefined || env.PORT === "" ? 8787 : Number(env.PORT);
  if (!Number.isInteger(port) || port <= 0 || port > 65535) throw new Error(`PORT must be an integer 1-65535`);
  const apiKey = env.W3W_API_KEY?.trim() ?? "";
  const apiSecret = env.W3W_API_SECRET?.trim() ?? "";
  if ((apiKey === "") !== (apiSecret === "")) throw new Error("set both W3W_API_KEY and W3W_API_SECRET, or neither");
  const keeperRaw = env.KEEPER_PK?.trim() ?? "";
  return {
    port,
    registry,
    rpcUrls: parseRpcUrls(env.BSC_RPC_URLS),
    logsRpcUrl: env.BSC_LOGS_RPC_URL?.trim() || DEFAULT_LOGS_RPC_URL,
    w3w: apiKey === "" ? null : { apiKey, apiSecret },
    keeperKey: keeperRaw === "" ? null : (keySchema.parse(keeperRaw) as `0x${string}`),
    telemetryFile: env.DEVEX_TELEMETRY_FILE?.trim() || null,
    corsOrigins: (env.CORS_ORIGINS ?? "*")
      .split(",")
      .map((origin) => origin.trim())
      .filter((origin) => origin.length > 0),
  };
}
```

import type { W3wClient } from "@pop/core";
import type { PublicClient } from "viem";
import type { ServerConfig } from "./config";

/** Chain reads the server needs (a viem PublicClient or a test fake). */
export type ChainClient = Pick<PublicClient, "readContract" | "getBlockNumber">;

/** The getLogs-capable RPC (publicnode by default; the official dataseed refuses eth_getLogs). */
export type LogsClient = Pick<PublicClient, "getLogs" | "getBlockNumber" | "getCode">;

export interface KeeperStatus {
  enabled: boolean;
  address: `0x${string}` | null;
  recorded: number;
  lastError: string | null;
}

export interface ServerDeps {
  config: ServerConfig;
  w3w: W3wClient | null;
  chain: ChainClient;
  logs: LogsClient;
  now: () => number;
  keeperStatus: () => KeeperStatus;
}

/** Thrown by W3W-backed routes when the server runs without API keys → HTTP 503. */
export class W3wNotConfiguredError extends Error {
  constructor() {
    super("the Binance Web3 API is not configured on this server");
    this.name = "W3wNotConfiguredError";
  }
}

export function requireW3w(deps: ServerDeps): W3wClient {
  if (!deps.w3w) throw new W3wNotConfiguredError();
  return deps.w3w;
}

/** A deliberate client-facing failure: the root onError turns it into { error: { code, msg } } with `status`. */
export class HttpError extends Error {
  readonly status: 400 | 404 | 409 | 503;
  readonly code: string;

  constructor(status: 400 | 404 | 409 | 503, code: string, message: string) {
    super(message);
    this.name = "HttpError";
    this.status = status;
    this.code = code;
  }
}

```ts
import { BSC_CHAIN_ID, W3W_ERROR_HINTS, W3wApiError } from "@pop/core";
import { Hono } from "hono";
import { cors } from "hono/cors";
import { ZodError } from "zod";
import { W3wNotConfiguredError, type ServerDeps } from "./deps";
import { HttpError } from "./errors";

/** A route module: returns a Hono sub-app mounted under /api. */
export type RouteModule = (deps: ServerDeps) => Hono;

/**
 * The API: GET /api/health plus every route module under /api, behind CORS, with one error mapper:
 * HttpError → its status · no API keys → 503 · Binance API failure → 502 with the verbatim code/msg ·
 * bad input → 400 · anything else → 500 (logged, never echoed).
 */
export function createApp(deps: ServerDeps, routes: readonly RouteModule[] = []): Hono {
  const app = new Hono();
  app.use("/api/*", cors({ origin: deps.config.corsOrigins.includes("*") ? "*" : deps.config.corsOrigins }));
  app.get("/api/health", (c) =>
    c.json({ ok: true, chainId: BSC_CHAIN_ID, registry: deps.config.registry, w3w: deps.w3w !== null, keeper: deps.keeperStatus() }),
  );
  for (const route of routes) app.route("/api", route(deps));
  app.notFound((c) => c.json({ error: { code: "NOT_FOUND", msg: "no such route" } }, 404));
  app.onError((error, c) => {
    if (error instanceof HttpError) return c.json({ error: { code: error.code, msg: error.message } }, error.status);
    if (error instanceof W3wNotConfiguredError) return c.json({ error: { code: "W3W_NOT_CONFIGURED", msg: error.message } }, 503);
    if (error instanceof W3wApiError) {
      return c.json({ error: { code: error.code, msg: error.msg, hint: W3W_ERROR_HINTS[error.code] ?? null } }, 502);
    }
    if (error instanceof ZodError) {
      const msg = error.issues.map((issue) => `${issue.path.join(".") || "body"}: ${issue.message}`).join("; ");
      return c.json({ error: { code: "BAD_REQUEST", msg } }, 400);
    }
    if (error instanceof SyntaxError) return c.json({ error: { code: "BAD_REQUEST", msg: "request body is not valid JSON" } }, 400);
    console.error("[server] unexpected error", error);
    return c.json({ error: { code: "INTERNAL", msg: "internal error" } }, 500);
  });
  return app;
}
```

export interface FakeChainState {
  calls: RawCall[];
  statuses: number[];
  allowance: bigint;
  blockNumber: bigint;
}

/** Answers readContract for callCount / getCalls (CallRegistry) and allowance (ERC-20) from `state`. */
export function fakeChain(state: FakeChainState): ChainClient {
  return {
    async readContract(request: { functionName: string; args?: readonly unknown[] }) {
      if (request.functionName === "callCount") return BigInt(state.calls.length);
      if (request.functionName === "getCalls") {
        const start = Number(request.args?.[0]);
        const count = Number(request.args?.[1]);
        return [state.calls.slice(start, start + count), state.statuses.slice(start, start + count)];
      }
      if (request.functionName === "allowance") return state.allowance;
      throw new Error(`fakeChain: unexpected ${request.functionName}`);
    },
    async getBlockNumber() {
      return state.blockNumber;
    },
  } as unknown as ChainClient;
}

export interface FakeLog {
  args: { from: string; to: string; value: bigint };
  transactionHash: string;
  blockNumber: bigint;
}

export function fakeLogs(logs: FakeLog[], codeByAddress: Record<string, string> = {}, latest = 1_000n): LogsClient & { requests: unknown[] } {
  const requests: unknown[] = [];
  return {
    requests,
    async getBlockNumber() {
      return latest;
    },
    async getLogs(request: unknown) {
      requests.push(request);
      return logs;
    },
    async getCode({ address }: { address: string }) {
      return codeByAddress[address.toLowerCase()] ?? undefined;
    },
  } as unknown as LogsClient & { requests: unknown[] };
}

export function makeDeps(
  overrides: {
    config?: Partial<ServerConfig>;
    w3w?: W3wClient | null;
    chain?: ChainClient;
    logs?: LogsClient;
    now?: () => number;
    keeper?: KeeperStatus;
  } = {},
): ServerDeps {
  return {
    config: testConfig(overrides.config),
    w3w: overrides.w3w === undefined ? createFakeW3wClient(() => null).client : overrides.w3w,
    chain: overrides.chain ?? fakeChain({ calls: [], statuses: [], allowance: 0n, blockNumber: 1_000n }),
    logs: overrides.logs ?? fakeLogs([]),
    now: overrides.now ?? (() => 1_791_000_000_000),
    keeperStatus: () => overrides.keeper ?? { enabled: false, address: null, recorded: 0, lastError: null },
  };
}

/** A fake W3W client answering by request path; unknown paths throw so tests notice unexpected calls. */
export function w3wByPath(answers: Record<string, (call: RecordedCall) => unknown>) {
  return createFakeW3wClient((call) => {
    const answer = answers[call.path];
    if (!answer) throw new Error(`unexpected W3W call ${call.method} ${call.path}`);
    return answer(call);
  });
}

/** Single-value cache: concurrent callers share one load; a failed load is not cached. */
export function createTtlCache<T>(ttlMs: number, now: () => number) {
  let value: { data: T; expires: number } | null = null;
  let pending: Promise<T> | null = null;
  return {
    async get(load: () => Promise<T>): Promise<T> {
      if (value !== null && now() < value.expires) return value.data;
      if (pending !== null) return pending;
      pending = load()
        .then((data) => {
          value = { data, expires: now() + ttlMs };
          return data;
        })
        .finally(() => {
          pending = null;
        });
      return pending;
    },
  };
}

/** Per-key cache with a TTL chosen per value (e.g. long for found evidence, short for "not yet"). */
export function createKeyedCache<K, T>(now: () => number) {
  const entries = new Map<K, { data: T; expires: number }>();
  return {
    async get(key: K, load: () => Promise<T>, ttlFor: (data: T) => number): Promise<T> {
      const hit = entries.get(key);
      if (hit && now() < hit.expires) return hit.data;
      const data = await load();
      entries.set(key, { data, expires: now() + ttlFor(data) });
      return data;
    },
  };
}

  } else {
    try {
      rwa = await getRwaTokens(deps.w3w);
    } catch (error) {
      if (!(error instanceof W3wApiError)) throw error;
      degraded = { code: error.code, msg: error.msg };
    }

      tokenToShareRatio: enriched?.tokenToShareRatio ?? null,
      priceUsd: enriched?.tokenPrice ?? null,
      status: enriched?.statusInfo ?? null,

  const token = getTokenInfo(body.token);
  if (!token) throw new HttpError(400, "UNKNOWN_TOKEN", "token is not in the tokenized-stock catalog");
  if (!isFollowable(token)) throw new HttpError(400, "NOT_FOLLOWABLE", "follows execute on bStocks only (Ondo is RFQ-only, xStocks are illiquid on BSC)");
  const amount = parseUnits18(body.usdtAmount);
  if (amount === 0n) throw new HttpError(400, "ZERO_AMOUNT", "usdtAmount must be greater than zero");
  const routes = await getQuote(requireW3w(deps), { fromToken: ADDRESSES.usdt, toToken: token.address, amount, userWalletAddress: body.wallet });
  const route = pickSwapRoute(routes);
  if (!route) throw new HttpError(409, "NO_SWAP_ROUTE", "no SWAP route right now (only RFQ, which closes outside US market hours)");
  return { route, amount };
}

function asUnsigned(tx: { from: string; to: string; data: string; value: string; gas?: string | null; gasPrice?: string | null }): UnsignedTx {
  return { from: tx.from, to: tx.to, data: tx.data, value: tx.value, gas: tx.gas ?? null, gasPrice: tx.gasPrice ?? null };
}

import { W3wApiError, formatUnits18, getTokenTrades, type CallView } from "@pop/core";
import { parseAbiItem } from "viem";
import type { ServerDeps } from "../deps";

export interface ExitEvidence {
  /** sale: Market API saw a sell · contract: tokens moved into a contract (likely a swap) · transfer: moved to a wallet. */
  kind: "sale" | "contract" | "transfer";
  txHash: string;
  /** Token amount, decimal string. */
  amount: string | null;
  timeMs: number | null;
  priceUsd: string | null;
  volumeUsd: string | null;
  dexName: string | null;
  counterparty: string | null;
  source: "market-api" | "rpc-logs";
}

const TRANSFER_EVENT = parseAbiItem("event Transfer(address indexed from, address indexed to, uint256 value)");

/** How far back the logs fallback looks: 400 blocks ≈ 3 minutes at 0.45 s blocks. */
export const LOG_LOOKBACK_BLOCKS = 400n;

/**
 * Where did the caller's tokens go? First the Market API (`trades` with `walletAddressFilter`, newest
 * sell at or after the post), then `Transfer(from = caller)` logs on the getLogs-capable RPC.
 */
export async function findExitEvidence(deps: Pick<ServerDeps, "w3w" | "logs">, call: CallView): Promise<ExitEvidence | null> {
  if (deps.w3w !== null) {
    try {
      const page = await getTokenTrades(deps.w3w, { token: call.token, wallets: [call.caller], limit: 20 });
      const sale = page.trades.find((trade) => trade.type.toLowerCase() === "sell" && trade.time >= call.postedAt * 1000);
      if (sale) {
        return {
          kind: "sale",
          txHash: sale.txHash,
          amount: sale.changedTokenInfo[0]?.amount ?? null,
          timeMs: sale.time,
          priceUsd: sale.price ?? null,
          volumeUsd: sale.volume ?? null,
          dexName: sale.dexName ?? null,
          counterparty: null,
          source: "market-api",
        };
      }
    } catch (error) {
      if (!(error instanceof W3wApiError)) throw error;
    }
  }

  const latest = await deps.logs.getBlockNumber();
  const floor = latest > LOG_LOOKBACK_BLOCKS ? latest - LOG_LOOKBACK_BLOCKS : 0n;
  const fromBlock = call.postedBlock > floor ? call.postedBlock : floor;
  const logs = await deps.logs.getLogs({ address: call.token, event: TRANSFER_EVENT, args: { from: call.caller }, fromBlock, toBlock: latest });
  const last = logs[logs.length - 1];
  if (!last || last.args.to === undefined || last.transactionHash === null) return null;
  const code = await deps.logs.getCode({ address: last.args.to });
  return {
    kind: code !== undefined && code !== "0x" ? "contract" : "transfer",
    txHash: last.transactionHash,
    amount: last.args.value === undefined ? null : formatUnits18(last.args.value, 6),
    timeMs: null,
    priceUsd: null,
    volumeUsd: null,
    dexName: null,
    counterparty: last.args.to,
    source: "rpc-logs",
  };
}

    );
    return c.json(response);

import {
  CALL_REGISTRY_ABI,
  SimulationFailedError,
  needsExitRecord,
  readCallsFrom,
  sendWithTransactionApi,
  type PipelineClient,
  type RegistryReader,
  type TxSigner,
  type W3wClient,
} from "@pop/core";
import { encodeFunctionData } from "viem";

export interface KeeperDeps {
  chain: RegistryReader & PipelineClient;
  w3w: W3wClient;
  signer: TxSigner;
  registry: `0x${string}`;
  onRecorded?: (callId: number, txHash: string) => void;
  /** callId -1 = the read itself failed. */
  onError?: (callId: number, message: string) => void;
}

export interface Keeper {
  /** One pass: record every open call that is no longer HOLDING and has no recorded exit. Never throws. */
  tick(): Promise<number[]>;
}

export function createKeeper(deps: KeeperDeps): Keeper {
  let running = false;
  return {
    async tick() {
      if (running) return [];
      running = true;
      const recorded: number[] = [];
      try {
        const calls = await readCallsFrom(deps.chain, deps.registry, 0);
        for (const call of calls.filter(needsExitRecord)) {
          try {
            const data = encodeFunctionData({ abi: CALL_REGISTRY_ABI, functionName: "recordExit", args: [BigInt(call.id)] });
            const result = await sendWithTransactionApi({ w3w: deps.w3w, client: deps.chain, signer: deps.signer, tx: { to: deps.registry, data }, mev: false });
            recorded.push(call.id);
            deps.onRecorded?.(call.id, result.txHash);
          } catch (error) {
            const message = error instanceof SimulationFailedError ? `simulation failed: ${error.reason}` : error instanceof Error ? error.message : String(error);
            deps.onError?.(call.id, message);
          }
        }
      } catch (error) {
        deps.onError?.(-1, error instanceof Error ? error.message : String(error));
      } finally {
        running = false;
      }
      return recorded;
    },
  };
}

/** Runs keeper.tick() every intervalMs (a tick still running is skipped). Returns a stop function. */
export function startKeeperLoop(keeper: Keeper, intervalMs = 1_000): () => void {
  const timer = setInterval(() => {
    void keeper.tick();
  }, intervalMs);
  return () => clearInterval(timer);
}

  const config = loadConfig(env);

  const w3w = config.w3w

import { useSyncExternalStore } from "react";

export type Route =
  | { name: "feed" }
  | { name: "call"; id: number }
  | { name: "post" }
  | { name: "how" }
  | { name: "not-found" };

/** Hash routes: #/ · #/calls/:id · #/post · #/how (no router dependency; works on any static host). */
export function parseHash(hash: string): Route {
  const path = hash.replace(/^#/, "") || "/";
  if (path === "/") return { name: "feed" };
  const call = /^\/calls\/(\d+)$/.exec(path);
  if (call) return { name: "call", id: Number(call[1]) };
  if (path === "/post") return { name: "post" };
  if (path === "/how") return { name: "how" };
  return { name: "not-found" };
}

export function hrefFor(route: Route): string {
  switch (route.name) {
    case "feed":
      return "#/";
    case "call":
      return `#/calls/${route.id}`;
    case "post":
      return "#/post";
    case "how":
      return "#/how";
    case "not-found":
      return "#/";
  }
}

function subscribe(onChange: () => void): () => void {
  window.addEventListener("hashchange", onChange);
  return () => window.removeEventListener("hashchange", onChange);
}

export function useRoute(): Route {
  const hash = useSyncExternalStore(subscribe, () => window.location.hash, () => "");
  return parseHash(hash);
}

  kind: "trimmed" | "exited" | "closed" | "exit-recorded";
}

/** Compares two reads of the same calls and reports what flipped between them (new calls are not flips). */
export function diffCalls(previous: readonly CallView[], next: readonly CallView[]): Flip[] {
  const before = new Map(previous.map((call) => [call.id, call]));
  const flips: Flip[] = [];
  for (const call of next) {
    const old = before.get(call.id);
    if (!old) continue;
    if (old.status !== call.status) {
      if (call.status === "EXITED") flips.push({ callId: call.id, from: old.status, to: call.status, kind: "exited" });
      else if (call.status === "TRIMMED") flips.push({ callId: call.id, from: old.status, to: call.status, kind: "trimmed" });
      else if (call.status === "CLOSED") flips.push({ callId: call.id, from: old.status, to: call.status, kind: "closed" });
    } else if (old.exitBlock === 0n && call.exitBlock > 0n) {
      flips.push({ callId: call.id, from: old.status, to: call.status, kind: "exit-recorded" });
    }
  }
  return flips;
}

export interface BlockPollState<T> {
  value: T | null;
  blockNumber: bigint | null;
  loading: boolean;
  /** Set after three failed reads in a row; the last good value stays on screen meanwhile. */
  error: string | null;
}

/**
 * Runs `read` on every new BSC block (1 s polling; blocks are ~0.45 s) and skips a block while a read
 * is still in flight. `key` identifies what is read: null disables polling, a new key restarts it.
 */
export function useBlockPoll<T>(
  client: Pick<PublicClient, "watchBlockNumber">,
  key: string | null,
  read: () => Promise<T>,
  onValue?: (value: T, previous: T | null) => void,
): BlockPollState<T> {
  const [state, setState] = useState<BlockPollState<T>>({ value: null, blockNumber: null, loading: key !== null, error: null });
  const readRef = useRef(read);
  readRef.current = read;
  const onValueRef = useRef(onValue);
  onValueRef.current = onValue;

  useEffect(() => {
    if (key === null) return;
    setState({ value: null, blockNumber: null, loading: true, error: null });
    let previous: T | null = null;
    let inFlight = false;
    let failures = 0;
    let cancelled = false;

    const refresh = async (blockNumber: bigint) => {
      if (inFlight) return;
      inFlight = true;
      try {
        const value = await readRef.current();
        if (cancelled) return;
        failures = 0;
        const before = previous;
        previous = value;
        setState({ value, blockNumber, loading: false, error: null });
        onValueRef.current?.(value, before);
      } catch (error) {
        failures += 1;
        if (!cancelled && failures >= 3) {
          setState((current) => ({ ...current, loading: false, error: error instanceof Error ? error.message : String(error) }));
        }
      } finally {
        inFlight = false;
      }
    };

    const unwatch = client.watchBlockNumber({
      emitOnBegin: true,
      pollingInterval: 1_000,
      onBlockNumber: (blockNumber) => void refresh(blockNumber),
      onError: (error) => setState((current) => ({ ...current, error: error.message })),
    });
    return () => {
      cancelled = true;
      unwatch();
    };
  }, [client, key]);

  return state;
}

  total: number;
  blockNumber: bigint | null;
  loading: boolean;
  error: string | null;

const NO_CALLS: CallView[] = [];

/** The newest `limit` calls (one getCalls eth_call per block, straight from public RPC) plus their flips. */
export function useLiveCalls(
  client: LiveCallsClient,
  registry: `0x${string}` | null,
  options: { limit?: number; onFlips?: (flips: Flip[], calls: CallView[]) => void } = {},
): LiveCallsState {

  const onFlips = useRef(options.onFlips);
  onFlips.current = options.onFlips;
  const poll = useBlockPoll(
    client,
    registry === null ? null : `${registry}:latest:${limit}`,
    () => readLatestCalls(client, registry as `0x${string}`, limit),

      if (previous === null) return;
      const flips = diffCalls(previous.calls, next.calls);
      if (flips.length > 0) onFlips.current?.(flips, next.calls);
    },
  );
  return {
    calls: poll.value?.calls ?? NO_CALLS,
    total: poll.value?.total ?? 0,
    blockNumber: poll.blockNumber,
    loading: poll.loading,
    error: poll.error,
  };

import type { CallView } from "@pop/core/browser";

export type BadgeTone = "holding" | "trimmed" | "exited" | "closed" | "closed-exited";

export interface Badge {
  label: string;
  detail: string;
  tone: BadgeTone;
}

/** The live badge: the registry's statusOf() for this call, read from BSC on every block. */
export function badgeFor(call: CallView): Badge {
  switch (call.status) {
    case "HOLDING":
      return { label: "HOLDING ✓", detail: "verified on-chain", tone: "holding" };
    case "TRIMMED":
      return { label: "TRIMMED", detail: "holds less than declared", tone: "trimmed" };
    case "EXITED":
      return { label: "EXITED", detail: "no longer holds it", tone: "exited" };
    case "CLOSED":
      return call.heldThrough
        ? { label: "CLOSED", detail: "held through ✓", tone: "closed" }
        : { label: "CLOSED", detail: "exited while open", tone: "closed-exited" };
  }
}

/** The sticky line: once an exit is recorded it stays, even if the caller buys back. */
export function exitLine(call: CallView): string | null {
  if (call.exitBlock === 0n) return null;
  const after = call.exitBlock - call.postedBlock;
  return `Exit recorded on-chain · block ${call.exitBlock.toString()} · ${after.toString()} blocks after post`;
}

/** True once the call has shown an exit: live TRIMMED/EXITED, a recorded exit, or closed without holding through. */
export function hasFlipped(call: CallView): boolean {
  return call.exitBlock > 0n || call.status === "TRIMMED" || call.status === "EXITED" || (call.status === "CLOSED" && !call.heldThrough);
}

import { useEffect, useState } from "react";
import type { ExitEvidence, TokenDto } from "../api/client";
import { companyName, links, shortAddress } from "../format";
import { useServices } from "../services";

export interface ExitEvidenceState {
  evidence: ExitEvidence | null;
  searching: boolean;
  error: string | null;
}

/** Asks the server where the caller's tokens went; retries until found (the Market API indexes with a lag). */
export function useExitEvidence(callId: number, enabled: boolean, options: { retryMs?: number; maxAttempts?: number } = {}): ExitEvidenceState {
  const { api } = useServices();
  const retryMs = options.retryMs ?? 4_000;
  const maxAttempts = options.maxAttempts ?? 15;
  const [state, setState] = useState<ExitEvidenceState>({ evidence: null, searching: enabled, error: null });
  useEffect(() => {
    if (!enabled) return;
    let cancelled = false;
    let timer: ReturnType<typeof setTimeout> | undefined;
    let attempts = 0;
    const attempt = async () => {
      attempts += 1;
      try {
        const response = await api.exit(callId);
        if (cancelled) return;
        if (response.evidence !== null) {
          setState({ evidence: response.evidence, searching: false, error: null });
          return;
        }
      } catch (error) {
        if (cancelled) return;
        setState({ evidence: null, searching: false, error: error instanceof Error ? error.message : String(error) });
        return;
      }
      if (attempts >= maxAttempts) setState({ evidence: null, searching: false, error: null });
      else timer = setTimeout(() => void attempt(), retryMs);
    };
    setState({ evidence: null, searching: true, error: null });
    void attempt();
    return () => {
      cancelled = true;
      if (timer !== undefined) clearTimeout(timer);
    };
  }, [api, callId, enabled, retryMs, maxAttempts]);
  return state;
}

export function evidenceText(evidence: ExitEvidence, token: TokenDto | undefined): string {
  const amount = evidence.amount ?? "";
  const symbol = token?.symbol ?? "tokens";
  switch (evidence.kind) {
    case "sale":
      return `sold ${amount} ${companyName(token)}${evidence.dexName ? ` · ${evidence.dexName}` : ""}${evidence.priceUsd ? ` @ $${evidence.priceUsd}` : ""}`;
    case "contract":
      return `moved ${amount} ${symbol} into a contract (likely a sale)`;
    case "transfer":
      return `transferred ${amount} ${symbol} to ${evidence.counterparty ? shortAddress(evidence.counterparty) : "another wallet"}`;
  }
}

export function ExitEvidenceLine({ callId, token, options }: { callId: number; token: TokenDto | undefined; options?: { retryMs?: number; maxAttempts?: number } }) {
  const state = useExitEvidence(callId, true, options);
  if (state.evidence) {
    return (
      <p className="evidence">
        {evidenceText(state.evidence, token)} ·{" "}
        <a href={links.tx(state.evidence.txHash)} target="_blank" rel="noreferrer">
          tx {shortAddress(state.evidence.txHash)}
        </a>
      </p>
    );
  }
  if (state.searching) return <p className="evidence muted">Looking for the exit transaction…</p>;
  if (state.error) return <p className="evidence muted">Exit evidence unavailable: {state.error}</p>;
  return <p className="evidence muted">Exit transaction not found in the recent blocks.</p>;
}

  request(args: { method: string; params?: unknown[] | Record<string, unknown> }): Promise<unknown>;
}

  uuid: string;
  name: string;
  icon: string;
  rdns: string;
}

export interface ProviderDetail {
  info: WalletInfo;
  provider: Eip1193Provider;
}

export const BSC_CHAIN_HEX = "0x38";

const BSC_CHAIN_PARAMS = {
  chainId: BSC_CHAIN_HEX,
  chainName: "BNB Smart Chain",
  nativeCurrency: { name: "BNB", symbol: "BNB", decimals: 18 },
  rpcUrls: ["https://bsc-dataseed.bnbchain.org"],
  blockExplorerUrls: ["https://bscscan.com"],
};

/** EIP-6963 discovery: ask every injected wallet to announce itself; fall back to window.ethereum. */
export function discoverProviders(target: Window = window, waitMs = 300): Promise<ProviderDetail[]> {
  return new Promise((resolve) => {
    const found = new Map<string, ProviderDetail>();
    const onAnnounce = (event: Event) => {
      const detail = (event as CustomEvent<ProviderDetail>).detail;
      if (detail?.info?.uuid && detail.provider) found.set(detail.info.uuid, detail);
    };
    target.addEventListener("eip6963:announceProvider", onAnnounce);
    target.dispatchEvent(new Event("eip6963:requestProvider"));
    setTimeout(() => {
      target.removeEventListener("eip6963:announceProvider", onAnnounce);
      const providers = [...found.values()];
      const legacy = (target as Window & { ethereum?: Eip1193Provider }).ethereum;
      if (providers.length === 0 && legacy) {
        providers.push({ info: { uuid: "legacy", name: "Browser wallet", icon: "", rdns: "legacy" }, provider: legacy });
      }
      resolve(providers);
    }, waitMs);
  });
}

/** Switches the wallet to BSC mainnet (adding the network when the wallet does not know it: error 4902). */
export async function ensureBsc(provider: Eip1193Provider): Promise<void> {
  const current = await provider.request({ method: "eth_chainId" });
  if (typeof current === "string" && current.toLowerCase() === BSC_CHAIN_HEX) return;
  try {
    await provider.request({ method: "wallet_switchEthereumChain", params: [{ chainId: BSC_CHAIN_HEX }] });
  } catch (error) {
    if ((error as { code?: number }).code !== 4902) throw error;
    await provider.request({ method: "wallet_addEthereumChain", params: [BSC_CHAIN_PARAMS] });
  }
}

export async function connectWallet(provider: Eip1193Provider): Promise<`0x${string}`> {
  const accounts = await provider.request({ method: "eth_requestAccounts" });
  const first = Array.isArray(accounts) ? accounts[0] : undefined;
  if (typeof first !== "string") throw new Error("the wallet returned no account");
  await ensureBsc(provider);
  return first.toLowerCase() as `0x${string}`;
}

export interface WalletTx {
  from: string;
  to: string;
  data: string;
  /** Wei, decimal string. */
  value?: string;
  /** Gas limit, decimal string. */
  gas?: string | null;
}

/** eth_sendTransaction on BSC; the wallet shows the transaction and signs it. Returns the tx hash. */
export async function sendTransaction(provider: Eip1193Provider, tx: WalletTx): Promise<`0x${string}`> {
  await ensureBsc(provider);
  const params: Record<string, string> = { from: tx.from, to: tx.to, data: tx.data, value: toHex(BigInt(tx.value ?? "0")) };
  if (tx.gas) params.gas = toHex(BigInt(tx.gas));
  const hash = await provider.request({ method: "eth_sendTransaction", params: [params] });
  if (typeof hash !== "string" || !hash.startsWith("0x")) throw new Error("the wallet returned no transaction hash");
  return hash as `0x${string}`;
}

/** Wallet errors carry { code, message }; 4001 is the user rejecting the request. */
export function walletErrorMessage(error: unknown): string {
  const { code, message } = (error ?? {}) as { code?: number; message?: string };
  if (code === 4001) return "Request rejected in the wallet.";
  return message ?? String(error);
}

import { CALL_REGISTRY_ABI } from "@pop/core/browser";
import { BaseError, ContractFunctionRevertedError, encodeFunctionData, type PublicClient } from "viem";
import { ApiError, type ApiClient } from "./api/client";

/** "Try the gate": a fixed address that holds no tokenized stock, posting 0.01 TSLAB. */
export const GATE_PROBE_ADDRESS = "0x000000000000000000000000000000000000c0de";
export const GATE_PROBE_SYMBOL = "TSLAB";
export const GATE_PROBE_AMOUNT = 10_000_000_000_000_000n;

export type DryRunVia = "Binance Transaction API simulate" | "BSC eth_call";

export interface DryRunResult {
  ok: boolean;
  /** The revert reason exactly as reported, e.g. "No position, no call." */
  reason: string | null;
  via: DryRunVia;
}

export interface PostArgs {
  registry: `0x${string}`;
  from: `0x${string}`;
  token: `0x${string}`;
  amount: bigint;
  note: string;
}

export function encodePost(token: `0x${string}`, amount: bigint, note: string): `0x${string}` {
  return encodeFunctionData({ abi: CALL_REGISTRY_ABI, functionName: "post", args: [token, amount, note] });
}

/** Human-readable revert: the require string, or CustomError(args) for the registry's custom errors. */
export function revertReason(error: unknown): string {
  if (error instanceof BaseError) {
    const reverted = error.walk((cause) => cause instanceof ContractFunctionRevertedError);
    if (reverted instanceof ContractFunctionRevertedError) {
      if (reverted.reason) return reverted.reason;
      if (reverted.data) return `${reverted.data.errorName}(${(reverted.data.args ?? []).map(String).join(", ")})`;
      return reverted.shortMessage;
    }
    return error.shortMessage;
  }
  return error instanceof Error ? error.message : String(error);
}

/**
 * Simulates post(token, amount, note) from `from` without signing anything: first through the server's
 * Transaction API simulate, then (server unreachable or Web3 API unavailable) straight against BSC RPC.
 */
export async function dryRunPost(api: ApiClient, chain: Pick<PublicClient, "simulateContract">, args: PostArgs): Promise<DryRunResult> {
  try {
    const simulation = await api.dryRun({ from: args.from, to: args.registry, data: encodePost(args.token, args.amount, args.note) });
    return {
      ok: simulation.success,
      reason: simulation.success ? null : (simulation.failReason ?? simulation.status),
      via: "Binance Transaction API simulate",
    };
  } catch (error) {
    if (!(error instanceof ApiError) && !(error instanceof TypeError)) throw error;
  }
  try {
    await chain.simulateContract({
      address: args.registry,
      abi: CALL_REGISTRY_ABI,
      functionName: "post",
      args: [args.token, args.amount, args.note],
      account: args.from,
    });
    return { ok: true, reason: null, via: "BSC eth_call" };
  } catch (error) {
    return { ok: false, reason: revertReason(error), via: "BSC eth_call" };
  }
}

export function loadWatched(storage: KeyValueStore | null): Set<number> {
  try {
    const parsed: unknown = JSON.parse(storage?.getItem(WATCH_KEY) ?? "[]");
    return new Set(Array.isArray(parsed) ? parsed.filter((id): id is number => Number.isInteger(id) && id >= 0) : []);
  } catch {
    return new Set();
  }
}

export function saveWatched(storage: KeyValueStore | null, ids: ReadonlySet<number>): void {
  try {
    storage?.setItem(WATCH_KEY, JSON.stringify([...ids].sort((a, b) => a - b)));
  } catch {
    // storage full or blocked: alerts still work for this tab
  }
}

/** Watched call ids, persisted per browser. Watching asks for Notification permission once. */
export function useWatched(storage: KeyValueStore | null = safeStorage()): { watched: ReadonlySet<number>; toggle: (callId: number) => void } {
  const [watched, setWatched] = useState<ReadonlySet<number>>(() => loadWatched(storage));
  const toggle = useCallback(
    (callId: number) => {
      setWatched((current) => {
        const next = new Set(current);
        if (next.has(callId)) next.delete(callId);
        else next.add(callId);
        saveWatched(storage, next);
        return next;
      });
      if (typeof Notification !== "undefined" && Notification.permission === "default") void Notification.requestPermission();
    },
    [storage],
  );
  return { watched, toggle };
}

export function flipMessage(flip: Flip, call: CallView | undefined, token: TokenDto | undefined): string {
  const subject = `Call #${flip.callId} · ${companyName(token)}`;
  switch (flip.kind) {
    case "exited":
      return `${subject}: the caller exited. Badge is now EXITED.`;
    case "trimmed":
      return `${subject}: the caller trimmed below the declared size.`;
    case "closed":
      return `${subject}: closed by the caller${call?.heldThrough ? " · held through ✓" : " · exited while open"}.`;
    case "exit-recorded":
      return `${subject}: exit recorded on-chain at block ${call?.exitBlock.toString() ?? "?"}.`;
  }
}

export function notify(title: string, body: string): void {
  if (typeof Notification === "undefined" || Notification.permission !== "granted") return;
  try {
    new Notification(title, { body });
  } catch {
    // some mobile browsers only allow notifications from a service worker
  }
}

export interface Toast {
  id: number;
  text: string;
  tone: "info" | "bad";
}

/** In-page toasts that dismiss themselves after `ttlMs`. */
export function useToasts(ttlMs = 8_000): { toasts: Toast[]; push: (text: string, tone?: Toast["tone"]) => void; dismiss: (id: number) => void } {
  const [toasts, setToasts] = useState<Toast[]>([]);
  const nextId = useRef(0);
  const timers = useRef(new Set<ReturnType<typeof setTimeout>>());
  const dismiss = useCallback((id: number) => setToasts((current) => current.filter((toast) => toast.id !== id)), []);
  const push = useCallback(
    (text: string, tone: Toast["tone"] = "info") => {
      nextId.current += 1;
      const id = nextId.current;
      setToasts((current) => [...current.slice(-4), { id, text, tone }]);
      const timer = setTimeout(() => {
        timers.current.delete(timer);
        dismiss(id);
      }, ttlMs);
      timers.current.add(timer);
    },
    [dismiss, ttlMs],
  );
  useEffect(() => {
    const pending = timers.current;
    return () => pending.forEach((timer) => clearTimeout(timer));
  }, []);
  return { toasts, push, dismiss };
}
