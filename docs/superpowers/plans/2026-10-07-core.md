# Core Implementation Plan

**Goal:** Build `@pop/core`, the shared TypeScript library every off-chain part uses: the signed Binance Web3 API client and typed wrappers, the CallRegistry ABI and readers, the token catalog, amount formatting, DevEx telemetry, and the simulate → sign → broadcast transaction pipeline.
**Architecture:** A pnpm 10 workspace package that exports TypeScript source (`"exports": { ".": "./src/index.ts", "./browser": "./src/browser.ts" }`), consumed through Vite, Vitest and `tsx` without a build step. Every network edge is injected (fetch, clock, viem client, signer) so each module is unit-tested against fixtures shaped exactly like the official OpenAPI responses; `@pop/core/browser` is the secret-free subset for the web app.
**Tech Stack:** Node ≥ 22.12, pnpm 10.34.6, TypeScript ~5.9.3 (strict, ESM, `moduleResolution: Bundler`), Vitest ^3.2.7, tsx ^4.23.15, viem ^2.57.3, zod ^3.25.76
**Source spec:** docs/superpowers/specs/2026-10-06-proof-of-position-design.md

Execution notes (apply to every task in this plan):

- Run every command from the repository root. Each `Run:` line is one command that works unchanged in PowerShell and Git Bash.
- Imports are extensionless relative paths (`./sign`), as `moduleResolution: Bundler` allows. Never use `as any`; when a test fake needs a cast, use `as unknown as T`.
- The Binance Web3 API response shapes used in fixtures come from the official OpenAPI 3.1.1 schema (asset SHA-256 prefix `4ae094ae23e96a96`, fetched 2026-10-07); do not "correct" field names from memory.
- Cross-plan dependencies are written `Task <subsystem>:<N>`. `pnpm-lock.yaml` changes are an allowed side effect of any `pnpm install`.

```ts
import { createHmac } from "node:crypto";

export interface W3wSignInput {
  secret: string;
  /** Exactly the X-OC-TIMESTAMP header value, ISO 8601 with milliseconds. */
  timestamp: string;
  method: string;
  /** Path as sent on the wire, including the /build prefix and the raw query string. */
  requestPath: string;
  /** Raw request body for POST; "" for GET. */
  body: string;
}

/**
 * X-OC-SIGN = Base64(HMAC-SHA256(timestamp + METHOD + requestPath + body, secret)).
 * The #1 cause of `40102 Invalid signature` is signing a path without the /build prefix, so it is enforced.
 */
export function signW3wRequest(input: W3wSignInput): string {
  if (!input.requestPath.startsWith("/build/")) {
    throw new Error(`requestPath must start with /build/: ${input.requestPath}`);
  }
  const preHash = `${input.timestamp}${input.method.toUpperCase()}${input.requestPath}${input.body}`;
  return createHmac("sha256", input.secret).update(preHash, "utf8").digest("base64");
}
```

import { W3W_BASE_URL } from "../chain/constants";
import { W3wApiError } from "./errors";
import { signW3wRequest } from "./sign";

export interface W3wTiming {
  method: "GET" | "POST";
  /** Path without the /build prefix, e.g. /api/v1/dex/market/rwa/tokens */
  endpoint: string;
  httpStatus: number;
  code: number | null;
  msg: string | null;
  ms: number;
}

export type QueryValue = string | number | boolean | undefined;

export interface W3wClientOptions {
  apiKey: string;
  apiSecret: string;
  baseUrl?: string;
  fetchImpl?: typeof fetch;
  /** Source of X-OC-TIMESTAMP. */
  now?: () => Date;
  /** Monotonic milliseconds for latency measurement. */
  clock?: () => number;
  sleep?: (ms: number) => Promise<void>;
  onTiming?: (timing: W3wTiming) => void;
  /** Retries on HTTP 429 (honouring Retry-After). Default 2. */
  maxRetries?: number;
  /** X-OC-RECV-WINDOW in ms. Default 10000 (API max 60000). */
  recvWindowMs?: number;
}

export interface W3wClient {
  get<T>(path: string, query: Record<string, QueryValue>, schema: z.ZodType<T, z.ZodTypeDef, unknown>): Promise<T>;
  post<T>(path: string, body: unknown, schema: z.ZodType<T, z.ZodTypeDef, unknown>): Promise<T>;
}

const envelopeSchema = z.object({
  code: z.union([z.number(), z.string()]),
  msg: z.string().nullish(),
  data: z.unknown(),
});

/** Query string exactly as sent and signed: skips undefined, encodes with encodeURIComponent, keeps order. */
export function buildQueryString(query: Record<string, QueryValue>): string {
  return Object.entries(query)
    .filter((entry): entry is [string, string | number | boolean] => entry[1] !== undefined)
    .map(([key, value]) => `${encodeURIComponent(key)}=${encodeURIComponent(String(value))}`)
    .join("&");
}

export function createW3wClient(options: W3wClientOptions): W3wClient {
  const baseUrl = options.baseUrl ?? W3W_BASE_URL;
  const fetchImpl = options.fetchImpl ?? fetch;
  const now = options.now ?? (() => new Date());
  const clock = options.clock ?? (() => performance.now());
  const sleep = options.sleep ?? ((ms: number) => new Promise<void>((resolve) => setTimeout(resolve, ms)));
  const maxRetries = options.maxRetries ?? 2;
  const recvWindowMs = options.recvWindowMs ?? 10_000;

  async function request<T>(
    method: "GET" | "POST",
    path: string,
    queryString: string,
    body: string | undefined,
    schema: z.ZodType<T, z.ZodTypeDef, unknown>,
  ): Promise<T> {
    const requestPath = `/build${path}${queryString ? `?${queryString}` : ""}`;
    for (let attempt = 0; ; attempt++) {
      const timestamp = now().toISOString();
      const headers: Record<string, string> = {
        "X-OC-APIKEY": options.apiKey,
        "X-OC-TIMESTAMP": timestamp,
        "X-OC-SIGN": signW3wRequest({ secret: options.apiSecret, timestamp, method, requestPath, body: body ?? "" }),
        "X-OC-RECV-WINDOW": String(recvWindowMs),
        Accept: "application/json",
      };
      if (body !== undefined) headers["Content-Type"] = "application/json";

      const started = clock();
      const response = await fetchImpl(`${baseUrl}${requestPath}`, { method, headers, body });
      const text = await response.text();
      const ms = Math.round(clock() - started);

      let envelope: z.infer<typeof envelopeSchema> | null = null;
      try {
        const parsed = envelopeSchema.safeParse(JSON.parse(text));
        envelope = parsed.success ? parsed.data : null;
      } catch {
        envelope = null;
      }
      const code = envelope ? Number(envelope.code) : null;
      const msg = envelope?.msg ?? null;
      options.onTiming?.({ method, endpoint: path, httpStatus: response.status, code, msg, ms });

      if (response.status === 429) {
        if (attempt < maxRetries) {
          const retryAfter = Number(response.headers.get("retry-after") ?? "1");
          await sleep((Number.isFinite(retryAfter) && retryAfter > 0 ? retryAfter : 1) * 1000);
          continue;
        }
        throw new W3wApiError({ code: 42900, msg: msg ?? "rate limited", httpStatus: 429, endpoint: path });
      }
      if (!envelope) {
        throw new W3wApiError({
          code: -1,
          msg: `non-JSON response: ${text.slice(0, 200)}`,
          httpStatus: response.status,
          endpoint: path,
        });
      }
      if (code !== 0 || !response.ok) {
        throw new W3wApiError({ code: code ?? -1, msg: msg ?? `HTTP ${response.status}`, httpStatus: response.status, endpoint: path });
      }
      return schema.parse(envelope.data);
    }
  }

  return {
    get: (path, query, schema) => request("GET", path, buildQueryString(query), undefined, schema),
    post: (path, body, schema) => request("POST", path, "", JSON.stringify(body), schema),
  };
}

export interface TelemetryOptions {
  /** JSONL file, e.g. devex/telemetry.jsonl (git-ignored; copy figures into DEVEX_LOG.md by hand). */
  filePath: string;
  /** Which process wrote the line: "server", "follower", "analyst", "demo". */

  now?: () => Date;

export interface TelemetryEvent {
  kind: string;
  [field: string]: string | number | boolean | null;
}

/** Appends one JSON line per event: { ts, service, ...event }. */
export function createEventRecorder(options: TelemetryOptions): (event: TelemetryEvent) => void {
  mkdirSync(dirname(options.filePath), { recursive: true });
  const now = options.now ?? (() => new Date());
  return (event) => {
    appendFileSync(options.filePath, `${JSON.stringify({ ts: now().toISOString(), service: options.service, ...event })}\n`);
  };
}

/** An onTiming callback for createW3wClient: one "w3w" line per API call (endpoint, status, code, msg, ms). */
export function createTelemetryRecorder(options: TelemetryOptions): (timing: W3wTiming) => void {
  const record = createEventRecorder(options);
  return (timing) => record({ kind: "w3w", ...timing });
}

const bySymbol = new Map<string, TokenInfo>(TOKEN_CATALOG.map((token) => [token.symbol.toLowerCase(), token]));

export const DEFAULT_WATCHLIST: readonly string[] = ["NVDAB", "TSLAB", "AAPLB", "SPYB"];

export function getTokenInfo(address: string): TokenInfo | undefined {
  return byAddress.get(address.toLowerCase());
}

export function getTokenBySymbol(symbol: string): TokenInfo | undefined {
  return bySymbol.get(symbol.toLowerCase());
}

export function requireTokenBySymbol(symbol: string): TokenInfo {
  const token = getTokenBySymbol(symbol);
  if (!token) throw new Error(`unknown token symbol ${symbol}`);
  return token;
}

export function tokensByIssuer(issuer: Issuer): TokenInfo[] {
  return TOKEN_CATALOG.filter((token) => token.issuer === issuer);
}

/** Case-insensitive search: exact symbol first, then symbol/ticker prefix matches, bStocks before others. */
export function findTokens(query: string, limit = 20): TokenInfo[] {
  const q = query.trim().toLowerCase();
  if (q === "") return [];
  const issuerRank: Record<Issuer, number> = { bStocks: 0, Ondo: 1, xStocks: 2 };
  return TOKEN_CATALOG.filter((token) => token.symbol.toLowerCase().startsWith(q) || token.ticker.toLowerCase().startsWith(q))
    .sort((a, b) => {
      const exactA = a.symbol.toLowerCase() === q ? 0 : 1;
      const exactB = b.symbol.toLowerCase() === q ? 0 : 1;
      return exactA - exactB || issuerRank[a.issuer] - issuerRank[b.issuer] || a.symbol.localeCompare(b.symbol);
    })
    .slice(0, limit);
}

/** Follow execution runs on bStocks only (spec §8.1): Ondo is RFQ-only and xStocks are illiquid on BSC. */
export function isFollowable(token: TokenInfo): boolean {
  return token.issuer === "bStocks";
}

```ts
const SCALE = 10n ** 18n;

/** "0.02" → 20000000000000000n. Accepts up to 18 decimals; rejects signs, exponents and empty input. */
export function parseUnits18(value: string): bigint {
  const trimmed = value.trim();
  if (!/^\d+(\.\d{1,18})?$/.test(trimmed) && !/^\.\d{1,18}$/.test(trimmed)) {
    throw new Error(`not a non-negative decimal with at most 18 places: "${value}"`);
  }
  const [whole = "0", fraction = ""] = trimmed.split(".");
  return BigInt(whole === "" ? "0" : whole) * SCALE + BigInt(fraction.padEnd(18, "0") || "0");
}

/** 20000000000000000n → "0.02"; rounds down to maxFractionDigits and trims trailing zeros. */
export function formatUnits18(raw: bigint, maxFractionDigits = 6): string {
  if (raw < 0n) return `-${formatUnits18(-raw, maxFractionDigits)}`;
  const whole = raw / SCALE;
  const fraction = (raw % SCALE).toString().padStart(18, "0").slice(0, maxFractionDigits).replace(/0+$/, "");
  return fraction === "" ? whole.toString() : `${whole}.${fraction}`;
}

/** Token amount (18 decimals) × tokenToShareRatio ("1.000778") → underlying shares, formatted. */
export function toShares(raw: bigint, tokenToShareRatio: string, maxFractionDigits = 4): string {
  return formatUnits18((raw * parseUnits18(tokenToShareRatio)) / SCALE, maxFractionDigits);
}

/** "187.43" USD price × 18-decimal token amount → USD value with 2 decimals. */
export function toUsd(raw: bigint, priceUsd: string): string {
  const cents = (raw * parseUnits18(priceUsd)) / SCALE / 10n ** 16n;
  return `${cents / 100n}.${(cents % 100n).toString().padStart(2, "0")}`;
}
```

import { CALL_REGISTRY_ABI } from "./abi.generated";
import { REPUTATION_REGISTRY_ABI } from "./abis";
import { toCallView, type CallView } from "./types";

/** The only client capability the readers need (real viem PublicClient or a test fake). */
export type RegistryReader = Pick<PublicClient, "readContract">;

export async function readCallCount(client: RegistryReader, registry: `0x${string}`): Promise<number> {
  const count = await client.readContract({ address: registry, abi: CALL_REGISTRY_ABI, functionName: "callCount" });
  return Number(count);
}

/** One getCalls page (oldest first), ids start at `start`. */
export async function readCallsPage(
  client: RegistryReader,
  registry: `0x${string}`,
  start: number,
  count: number,
): Promise<CallView[]> {
  const [calls, statuses] = await client.readContract({
    address: registry,
    abi: CALL_REGISTRY_ABI,
    functionName: "getCalls",
    args: [BigInt(start), BigInt(count)],
  });
  return calls.map((raw, index) => toCallView(start + index, raw, statuses[index] ?? 0));
}

/** The newest `limit` calls, newest first, plus the total count. */
export async function readLatestCalls(
  client: RegistryReader,
  registry: `0x${string}`,
  limit: number,
): Promise<{ total: number; calls: CallView[] }> {
  const total = await readCallCount(client, registry);
  if (total === 0) return { total, calls: [] };
  const start = Math.max(0, total - limit);
  const page = await readCallsPage(client, registry, start, total - start);
  return { total, calls: page.reverse() };
}

/** Every call from `fromId` on, oldest first, in pages of `pageSize`. */
export async function readCallsFrom(
  client: RegistryReader,
  registry: `0x${string}`,
  fromId: number,
  pageSize = 100,
): Promise<CallView[]> {
  const total = await readCallCount(client, registry);
  const calls: CallView[] = [];
  for (let start = fromId; start < total; start += pageSize) {
    calls.push(...(await readCallsPage(client, registry, start, Math.min(pageSize, total - start))));
  }
  return calls;
}

import type { LocalAccount, PublicClient, TransactionReceipt } from "viem";
import { BSC_CHAIN_ID } from "../chain/constants";
import type { W3wClient } from "../w3w/client";
import { broadcastTx, isSimulationSuccess, simulateTx, simulationFailure, type Simulation } from "../w3w/transaction";

/** A legacy (type 0) BSC transaction ready to sign. */
export interface LegacyTxRequest {
  chainId: number;
  nonce: number;
  to: `0x${string}`;
  data: `0x${string}`;
  value: bigint;
  gas: bigint;
  gasPrice: bigint;
}

/** Anything that can sign a raw transaction without broadcasting it. Keys never leave the signer. */
export interface TxSigner {
  address: `0x${string}`;
  signTransaction(tx: LegacyTxRequest): Promise<`0x${string}`>;
}

export type PipelineClient = Pick<PublicClient, "getTransactionCount" | "getGasPrice" | "estimateGas" | "waitForTransactionReceipt">;

export class SimulationFailedError extends Error {
  /** The Transaction API's failReason, e.g. "execution reverted: No position, no call." */
  readonly reason: string;

  constructor(reason: string) {
    super(`simulation failed: ${reason}`);
    this.name = "SimulationFailedError";
    this.reason = reason;
  }
}

export interface SendTxParams {
  w3w: W3wClient;
  client: PipelineClient;
  signer: TxSigner;
  tx: { to: `0x${string}`; data: `0x${string}`; value?: bigint; gas?: bigint; gasPrice?: bigint };
  /** Route the broadcast through the Transaction API's private mempool. */
  mev: boolean;
  /** Added on top of eth_estimateGas when tx.gas is not given. Default 20. */
  gasBufferPercent?: number;
  receiptTimeoutMs?: number;
  clock?: () => number;
}

export interface SendTxResult {
  txHash: `0x${string}`;
  orderId: string;
  receipt: TransactionReceipt;
  simulation: Simulation;
  timings: { simulateMs: number; broadcastMs: number; confirmMs: number };
}

/**
 * The project's only write path for keys it controls (demo personas, keeper, analyst):
 * Transaction API simulate → never sign on failure → sign locally → Transaction API broadcast → wait for the receipt.
 */
export async function sendWithTransactionApi(params: SendTxParams): Promise<SendTxResult> {
  const clock = params.clock ?? (() => performance.now());
  const from = params.signer.address;
  const value = params.tx.value ?? 0n;

  const started = clock();
  const simulation = await simulateTx(params.w3w, { from, to: params.tx.to, value: value.toString(), data: params.tx.data });
  if (!isSimulationSuccess(simulation)) throw new SimulationFailedError(simulationFailure(simulation));
  const simulated = clock();

  const nonce = await params.client.getTransactionCount({ address: from, blockTag: "pending" });
  const gasPrice = params.tx.gasPrice ?? (await params.client.getGasPrice());
  const buffer = BigInt(100 + (params.gasBufferPercent ?? 20));
  const gas =
    params.tx.gas ?? ((await params.client.estimateGas({ account: from, to: params.tx.to, data: params.tx.data, value })) * buffer) / 100n;
  const signedTransaction = await params.signer.signTransaction({
    chainId: BSC_CHAIN_ID,
    nonce,
    to: params.tx.to,
    data: params.tx.data,
    value,
    gas,
    gasPrice,
  });
  const { orderId, txHash } = await broadcastTx(params.w3w, { signedTransaction, address: from, enableMevProtection: params.mev });
  const broadcast = clock();

  const hash = txHash as `0x${string}`;
  const receipt = await params.client.waitForTransactionReceipt({ hash, timeout: params.receiptTimeoutMs ?? 90_000 });
  const confirmed = clock();
  if (receipt.status !== "success") throw new Error(`transaction ${hash} reverted on chain`);
  return {
    txHash: hash,
    orderId,
    receipt,
    simulation,
    timings: {
      simulateMs: Math.round(simulated - started),
      broadcastMs: Math.round(broadcast - simulated),
      confirmMs: Math.round(confirmed - broadcast),
    },

/** Follows runtime (non-type) relative imports/re-exports from `entry` and returns every reachable file. */
function runtimeImportGraph(entry: string): Map<string, string> {
  const files = new Map<string, string>();
  const queue = [resolve(srcDir, entry)];
  while (queue.length > 0) {
    const file = queue.pop() as string;
    if (files.has(file)) continue;
    const source = readFileSync(file, "utf8");
    files.set(file, source);
    const pattern = /^(?:import|export)\s+(?!type\b)[^;]*?from\s+"(\.{1,2}\/[^"]+)"/gm;
    for (const match of source.matchAll(pattern)) {
      const target = match[1] as string;
      queue.push(resolve(dirname(file), `${target}.ts`));
    }
  }
  return files;
}
